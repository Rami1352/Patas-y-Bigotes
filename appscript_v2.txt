// Normaliza un nombre de servicio al canónico del sistema
function normalizeServicio(raw) {
  var s = String(raw || '').trim().toUpperCase().replace(/[\s\-]+/g, '_');
  if (s.indexOf('DESPARASIT') !== -1) return 'DESPARASITARIOS';   // DESPARASITARIO → DESPARASITARIOS
  if (s.indexOf('CUIDADO') !== -1) return 'CUIDADO_DOMICILIO'; // CUIDADO DOMICILIO / CUIDADODOMICILIO
  if (s.indexOf('ALIMENTO') !== -1) return 'ALIMENTOS';
  if (s.indexOf('PASEO') !== -1) return 'PASEOS';
  return s;
}

// Normaliza la celda entera de Servicios_Activos
// Soporta separadores: coma, punto y coma, pipe, barra
function normalizeServiciosActivos(cellValue) {
  var raw = String(cellValue || '').trim().toUpperCase();
  if (!raw) return '';
  var parts = raw.split(/[,;|\/\\]+/);
  var result = [];
  for (var i = 0; i < parts.length; i++) {
    var norm = normalizeServicio(parts[i]);
    if (norm) result.push(norm);
  }
  return result.join(',');
}

function extractPhotoUrl(raw) {
  var s = String(raw || '').trim();
  if (!s) return '';

  var m = s.match(/=IMAGE\(\s*"([^"]+)"/i);
  if (m) s = m[1];

  var fileId = '';

  var mId = s.match(/[?&]id=([a-zA-Z0-9_-]{10,})/);
  if (mId) fileId = mId[1];

  if (!fileId) {
    var mPath = s.match(/\/file\/d\/([a-zA-Z0-9_-]{10,})/);
    if (mPath) fileId = mPath[1];
  }

  if (!fileId) {
    var mUgc = s.match(/googleusercontent\.com\/d\/([a-zA-Z0-9_-]{10,})/);
    if (mUgc) fileId = mUgc[1];
  }

  if (fileId) {
    return 'https://lh3.googleusercontent.com/d/' + fileId;
  }

  return s;
}

// ═══════════════════════════════════════════════════════════════
//  parsePrecioCLP — convierte el texto TAL COMO SE VE en la celda de
//  una columna de precio (ej. "$ 84.000", "84.000,50") a un NÚMERO
//  limpio en JavaScript, sin símbolo de moneda ni separador de miles.
// ═══════════════════════════════════════════════════════════════
function parsePrecioCLP(raw) {
  var s = String(raw == null ? '' : raw).trim();
  if (!s) return 0;

  s = s.replace(/[^0-9.,]/g, '');
  if (!s) return 0;

  if (s.indexOf(',') !== -1) {
    s = s.replace(/\./g, '').replace(',', '.');
  } else {
    s = s.replace(/\./g, '');
  }

  var n = parseFloat(s);
  return isNaN(n) ? 0 : n;
}

//  Para no pagar una llamada de red por request, el resultado se cachea

// Se leen de Propiedades del Script (Configuración → Propiedades del script).
// La Web API Key de Firebase es pública por diseño (viaja en el frontend),
// pero se deja configurable para no hardcodearla acá.
function getFirebaseApiKey() {
  return PropertiesService.getScriptProperties().getProperty('FIREBASE_API_KEY') || '';
}

// AUTH_ENFORCE — interruptor de la migración (sección 14 del plan).
//   false → si el request NO trae idToken, se responde como antes
//           (compatibilidad con el frontend viejo mientras se despliega).
//           ⚠️ Con false la fuga sigue abierta para quien no mande token.
//   true  → sin idToken válido no se devuelve ningún dato privado.
// Pasar a true recién cuando el frontend ya mande idToken en todos los
// fetch. Se puede cambiar sin editar código creando la propiedad de
// script AUTH_ENFORCE = "true".
function isAuthEnforced() {
  var p = PropertiesService.getScriptProperties().getProperty('AUTH_ENFORCE');
  return String(p).toLowerCase() === 'true';
}

// ── Verifica el idToken contra Google y devuelve { uid, email } o null ──
function verifyFirebaseToken(idToken) {
  if (!idToken) return null;
  var apiKey = getFirebaseApiKey();
  if (!apiKey) {
    Logger.log('[AUTH] Falta la propiedad de script FIREBASE_API_KEY.');
    return null;
  }

  var cache = CacheService.getScriptCache();
  var cacheKey = 'fbtok_' + Utilities.base64Encode(
    Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, idToken)
  );
  var hit = cache.get(cacheKey);
  if (hit) {
    try { return JSON.parse(hit); } catch (err) { /* cae al lookup */ }
  }

  try {
    var res = UrlFetchApp.fetch(
      'https://identitytoolkit.googleapis.com/v1/accounts:lookup?key=' + encodeURIComponent(apiKey),
      {
        method: 'post',
        contentType: 'application/json',
        payload: JSON.stringify({ idToken: idToken }),
        muteHttpExceptions: true
      }
    );
    if (res.getResponseCode() !== 200) return null; // vencido, forjado u otro proyecto

    var users = (JSON.parse(res.getContentText()) || {}).users || [];
    if (!users.length || !users[0].localId) return null;

    var out = {
      uid: String(users[0].localId),
      email: String(users[0].email || '').trim().toLowerCase()
    };
    // TTL corto: el token de Firebase dura 1 h; 5 min es margen de sobra.
    cache.put(cacheKey, JSON.stringify(out), 300);
    return out;
  } catch (err) {
    Logger.log('[AUTH] Error verificando token: ' + err);
    return null;
  }
}

// ── Firebase_UID → fila de la hoja Clientes ──
function resolveClienteFromUid(uid) {
  if (!uid) return null;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Clientes') || ss.getSheetByName('CLIENTES');
  if (!sheet) return null;

  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return null;

  var map = {};
  for (var c = 0; c < data[0].length; c++) {
    var norm = String(data[0][c] || '').trim().toLowerCase().replace(/[\s_]/g, '');
    if (norm) map[norm] = c;
  }
  if (!map.hasOwnProperty('firebaseuid')) return null;

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][map['firebaseuid']] || '').trim() === String(uid).trim()) {
      return {
        clienteId: map.hasOwnProperty('clienteid') ? String(data[i][map['clienteid']] || '').trim() : '',
        email: map.hasOwnProperty('email') ? String(data[i][map['email']] || '').trim().toLowerCase() : '',
        rol: (map.hasOwnProperty('rol') ? String(data[i][map['rol']] || '') : '').trim().toLowerCase() || 'usuario',
        estado: (map.hasOwnProperty('estado') ? String(data[i][map['estado']] || '') : '').trim().toUpperCase(),
        rowNumber: i + 1
      };
    }
  }
  return null;
}

// ── Contexto de autorización de un request ──
//  Devuelve:
//   { mode:'legacy' }                        → sin token y AUTH_ENFORCE=false
//   { mode:'denied' }                        → hace falta token y no hay/es inválido
//   { mode:'auth', cliente:{…}, isAdmin:… }  → identidad verificada
//
//  ⚠️ El `mail` / `clienteId` que mande el cliente NO se usa nunca para
//  decidir qué devolver: la identidad sale solo del token verificado.
function getAuthContext(e) {
  var p = (e && e.parameter) ? e.parameter : {};
  var idToken = p.idToken || p.idtoken || p.token || '';

  if (!idToken) return isAuthEnforced() ? { mode: 'denied' } : { mode: 'legacy' };

  var tok = verifyFirebaseToken(idToken);
  if (!tok) return { mode: 'denied' };

  var cliente = resolveClienteFromUid(tok.uid);
  if (!cliente) {
    // Token válido pero todavía sin vincular en la hoja Clientes.
    return { mode: 'auth', cliente: { clienteId: '', email: tok.email, rol: 'usuario', estado: 'PENDIENTE' }, isAdmin: false, uid: tok.uid };
  }
  // El email de referencia sale del token, no de la planilla.
  if (!cliente.email) cliente.email = tok.email;
  return { mode: 'auth', cliente: cliente, isAdmin: cliente.rol === 'admin', uid: tok.uid };
}

// ── Filtra filas ya materializadas según el contexto ──
//  Match por CLIENTE_ID cuando la fila lo tiene cargado; si está vacío
//  (migración a medias), cae a comparar Mail contra el email del token.
function filterRowsForContext(rows, ctx) {
  if (!ctx || ctx.mode === 'legacy') return rows;      // compatibilidad temporal
  if (ctx.mode === 'denied') return [];
  if (ctx.isAdmin) return rows;

  var cid = String(ctx.cliente.clienteId || '').trim();
  var email = String(ctx.cliente.email || '').trim().toLowerCase();

  return (rows || []).filter(function (r) {
    var rowCid = '';
    var rowMail = '';
    for (var k in r) {
      if (!r.hasOwnProperty(k)) continue;
      var nk = String(k).trim().toLowerCase().replace(/[\s_]/g, '');
      if (nk === 'clienteid') rowCid = String(r[k] || '').trim();
      if (nk === 'mail' || nk === 'email') rowMail = String(r[k] || '').trim().toLowerCase();
    }
    if (cid && rowCid) return rowCid === cid;
    return email !== '' && rowMail === email;
  });
}

// ── ¿La fila con ese Id pertenece al cliente autenticado? ──
function rowBelongsToContext(sheetName, id, ctx) {
  if (!id) return false;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return false;

  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return false;

  var map = {};
  for (var c = 0; c < data[0].length; c++) {
    var norm = String(data[0][c] || '').trim().toLowerCase()
      .normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/[\s_]/g, '');
    // Primera aparición gana: si CLIENTE_ID estuviera duplicada, vale la
    // de más a la izquierda (la que completa la migración).
    if (norm && !map.hasOwnProperty(norm)) map[norm] = c;
  }
  if (!map.hasOwnProperty('id')) return false;

  var cid = String(ctx.cliente.clienteId || '').trim();
  var email = String(ctx.cliente.email || '').trim().toLowerCase();

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][map['id']] || '').trim() !== String(id).trim()) continue;

    var rowCid = map.hasOwnProperty('clienteid') ? String(data[i][map['clienteid']] || '').trim() : '';
    var rowMail = map.hasOwnProperty('mail') ? String(data[i][map['mail']] || '').trim().toLowerCase() : '';

    if (cid && rowCid) return rowCid === cid;      // criterio fuerte
    return email !== '' && rowMail === email;      // fallback de migración
  }
  return false; // Id inexistente
}

// ── Punto de control único, compartido por doGet y doPost.
//    Recibe el objeto de parámetros (e.parameter en GET, el body parseado
//    en POST) y devuelve null si se puede seguir, o un string con el
//    motivo del rechazo. Muta `p` para forzar la identidad en las altas.
//
//    Se comparte a propósito: doPost expone EXACTAMENTE las mismas 12
//    acciones de escritura que doGet, así que protegerlas por separado
//    sería dejar la puerta de atrás abierta. ──
function checkWriteAuth(p) {
  p = p || {};
  var action = p.action || '';
  if (!WRITE_ACTION_SHEET.hasOwnProperty(action)) return null; // no es escritura

  var ctx = getAuthContext({ parameter: p });

  // Modo migración: sin token se responde como antes (no rompe el portal
  // viejo). ⚠️ Mientras AUTH_ENFORCE sea false esto sigue siendo un hueco.
  if (ctx.mode === 'legacy') return null;

  if (ctx.mode === 'denied') return 'No autorizado: falta un idToken válido.';

  var agendaWriteActions = ['addPaseo', 'updatePaseo', 'deletePaseo', 'duplicatePaseo'];
  if (agendaWriteActions.indexOf(action) !== -1 && !ctx.isAdmin) {
    return 'No autorizado: se requiere rol de administrador para modificar la agenda.';
  }

  if (ctx.isAdmin) return null; // el admin opera sobre cualquier registro

  // Cuenta verificada por Firebase pero todavía sin vincular a un cliente
  // y sin email utilizable: no hay forma de saber qué le pertenece.
  if (!ctx.cliente.clienteId && !ctx.cliente.email) {
    return 'Tu cuenta todavía no está vinculada. Contactá al administrador.';
  }

  // 1) ALTAS: se fuerza la identidad del token y se descarta lo que haya
  //    mandado el cliente, así nadie crea registros a nombre de otro.
  if (action.indexOf('add') === 0) {
    p.mail = ctx.cliente.email;
    if (ctx.cliente.clienteId) p.clienteId = ctx.cliente.clienteId;
    return null;
  }

  // 2) updatePeso / updateCumple: se identifican por mail, no por id →
  //    se fuerza el mail. Mismo criterio para ambas: si updateCumple no
  //    entrara acá, caería en el bloque 3 (MODIFICACIÓN/BORRADO) que
  //    exige un p.id propio, y esta acción nunca manda un Id.
  if (action === 'updatePeso' || action === 'updateCumple') {
    p.mail = ctx.cliente.email;
    return null;
  }

  // 3) MODIFICACIÓN / BORRADO: la fila apuntada tiene que ser suya.
  var sheetName = WRITE_ACTION_SHEET[action];
  if (!sheetName) return null;
  if (!rowBelongsToContext(sheetName, p.id, ctx)) {
    return 'No autorizado: ese registro no te pertenece.';
  }
  return null;
}

      }

      if (!mail) {
        for (var ci2 = 0; ci2 < row.length; ci2++) {
          var v = row[ci2].toString().trim();
          if (v.indexOf('@') !== -1 && v.indexOf('.') !== -1) { mail = v.toLowerCase(); break; }
        }
      }

      var walkAdded = false;
      var dayNames = ['Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado', 'Domingo'];

      for (var di = 0; di < dayNames.length; di++) {
        var dayName = dayNames[di];
        var colIdx = dayToColIndex[dayName];
        if (colIdx === -1) continue;
        var cellVal = row[colIdx].toString().trim();
        if (!cellVal) continue;

        var parts = cellVal.split(/\s*[-–a]\s*/i);
        var iniT = fixTime(parts[0] || '');
        var finT = fixTime(parts[1] || '');

        if (iniT && !finT) {
          var h2 = parseInt(iniT.split(':')[0], 10);
          var m2 = iniT.split(':')[1] || '00';
          finT = ((h2 + 1) % 24).toString().padStart(2, '0') + ':' + m2;
        }

        rows.push(makeRow(nombre, dueno, barrio, raza, sexo, cumple, lote, costo, dayName, iniT, finT, mail, rol, servicios, peso, foto, clienteIdRow, paseoIdRow));
        walkAdded = true;
      }

      if (!walkAdded) {
        var diaVal = (idx.dia !== -1) ? row[idx.dia].toString().trim() : '';
        var inicioVal = (idx.inicio !== -1) ? row[idx.inicio].toString().trim() : '';
        var finVal = (idx.fin !== -1) ? row[idx.fin].toString().trim() : '';

        if ((!inicioVal || !finVal) && idx.horario !== -1) {
          var hp = row[idx.horario].toString().trim().split(/\s*[-–a]\s*/i);
          if (hp.length >= 2) { inicioVal = hp[0].trim(); finVal = hp[1].trim(); }
        }

        inicioVal = fixTime(inicioVal);
        finVal = fixTime(finVal);

        if (diaVal) {
          var splitDays = diaVal.split(/[,;\/]+/);
          for (var sd = 0; sd < splitDays.length; sd++) {
            var sDay = splitDays[sd].trim();
            if (!sDay) continue;
            var dNorm = sDay.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
            var finalDay = sDay;
            if (dNorm === 'l' || dNorm.startsWith('lun')) finalDay = 'Lunes';
            if (dNorm === 'm' || dNorm === 'mar') finalDay = 'Martes';
            if (dNorm === 'mi' || dNorm === 'x' || dNorm.startsWith('mie')) finalDay = 'Miércoles';
            if (dNorm === 'j' || dNorm.startsWith('jue')) finalDay = 'Jueves';
            if (dNorm === 'v' || dNorm.startsWith('vie')) finalDay = 'Viernes';
            if (dNorm === 's' || dNorm.startsWith('sab')) finalDay = 'Sábado';
            if (dNorm === 'd' || dNorm.startsWith('dom')) finalDay = 'Domingo';
            rows.push(makeRow(nombre, dueno, barrio, raza, sexo, cumple, lote, costo, finalDay, inicioVal, finVal, mail, rol, servicios, peso, foto, clienteIdRow, paseoIdRow));
            walkAdded = true;
          }
        }

        if (!walkAdded) {
          rows.push(makeRow(nombre, dueno, barrio, raza, sexo, cumple, lote, costo, '', '', '', mail, rol, servicios, peso, foto, clienteIdRow, paseoIdRow));
        }
      }
    }

    // Mascota Paseo: se recorta a las mascotas del cliente autenticado.
    // Las filas se arman con makeRow(), que ya expone 'mail'; si la hoja
    // tiene CLIENTE_ID cargado, filterRowsForContext lo prioriza.
    return returnJSONP(filterRowsForContext(rows, getAuthContext(e)), e);

  } catch (error) {
    return returnJSONP([{ error: error.toString() }], e);
  }
}

function getOrCreateSheet(name, headers) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
  }
  return sheet;
}

function getHeaderMap(sheet) {
  var lastCol = sheet.getLastColumn();
  var headerRow = sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0];
  var map = {};
  for (var i = 0; i < headerRow.length; i++) {
    var norm = String(headerRow[i] || '')
      .trim().toLowerCase()
      .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
      .replace(/[\s_]/g, '');
    if (norm) map[norm] = i; // 0-based
  }
  return map;
}

// ═══════════════════════════════════════════════════════════════
//  getRegistroWidth — Devuelve cuántas columnas ocupa HOY el bloque de
//  REGISTRO de una hoja (Alimentacion / Desparasitarios), leyéndolo de
//  los headers reales en vez de confiar en una constante fija.
//
//  POR QUÉ EXISTE (fix v23):
//  COLUMNAS_REGISTRO_ALIMENTACION (10) y COLUMNAS_REGISTRO_DESPARASITACION
//  (7) asumían un orden de columnas que YA NO ES EL DE LA PLANILLA: al
//  insertar CODIGO_MASCOTA / CLIENTE_ID en el medio (columna C), todo el
//  registro se corrió una posición a la derecha y la columna Id quedó
//  FUERA del rango que esas constantes cubren:
//
//    Alimentacion   → Id pasó de J(10) a K(11), pero la constante decía 10
//    Desparasitarios→ Id pasó de G(7)  a H(8),  pero la constante decía 7
//
//  Consecuencia real (bug silencioso, sin excepción):
//   1) appendRowByHeaderInFirstEmptySlot* descarta cualquier valor cuyo
//      índice de header sea >= colsToWrite (ver el guard `headerMap[key]
//      < colsToWrite`). Con el corrimiento, ESE valor descartado es el
//      Id → cada alta nueva se guardaba SIN Id.
//   2) handleDelete* limpiaba solo hasta la constante, dejando el Id
//      viejo intacto una columna más a la derecha → al reutilizarse ese
//      hueco, el registro nuevo heredaba el Id del registro borrado.
//   Sin Id (o con un Id ajeno), editar/borrar desde el portal apunta al
//   registro equivocado o directamente no encuentra nada.
//
//  El registro siempre termina en la columna Id (el catálogo arranca
//  después), así que derivar el ancho como índice(Id)+1 es correcto sin
//  importar cuántas columnas se agreguen antes. Si la hoja todavía no
//  tiene Id (hoja recién creada), se cae al valor de la constante.
//
//  NOTA v23 (columnas de pago): las 5 columnas nuevas de
//  "Desparasitarios" (Metodo_Pago, Alias_Pago, Estado_Pago,
//  Referencia_Pago, Numero_Pedido) se agregan DESPUÉS de Id en los
//  headers, así que getRegistroWidth (que corta justo en Id) sigue
//  devolviendo lo mismo que antes para esa hoja: no las incluye en el
//  bloque de registro/catálogo que usa este cálculo. Se escriben aparte,
//  con sus propios índices de columna, en handleAddDesparasitario.
//
//  NOTA v25: mismo criterio aplicado ahora a 'Alimentacion' — las 5
//  columnas de pago nuevas de esa hoja también van DESPUÉS de Id, así
//  que esta función sigue sin cambios: corta en Id igual que siempre,
//  para ambas hojas.
// ═══════════════════════════════════════════════════════════════
function getRegistroWidth(sheet, fallbackConst) {
  var lastCol = sheet.getLastColumn();
  try {
    var map = getHeaderMap(sheet);
    if (map.hasOwnProperty('id')) {
      return Math.min(map['id'] + 1, lastCol); // registro = A .. Id inclusive
    }
  } catch (err) {
    // Si algo falla leyendo headers, se usa el comportamiento anterior.
  }
  return Math.min(fallbackConst, lastCol);
}

function appendRowByHeader(sheet, headerMap, values) {
  var lastCol = sheet.getLastColumn();
  var row = new Array(lastCol).fill('');
  for (var key in values) {
    if (!values.hasOwnProperty(key)) continue;
    if (headerMap.hasOwnProperty(key)) {
      row[headerMap[key]] = values[key];
    }
  }
  sheet.appendRow(row);
  return sheet.getLastRow();
}

// ═══════════════════════════════════════════════════════════════
//  findFirstEmptyRowInAlimentacionRegistro — Busca, en la hoja
//  "Alimentacion", la primera fila (empezando en la 2, justo debajo
//  del header) cuyo bloque de columnas A-J (registro; ver
//  COLUMNAS_REGISTRO_ALIMENTACION) esté completamente vacío. Ignora
//  por completo si esa misma fila tiene contenido de catálogo en K en
//  adelante (Nombre_alimentos_segun_link, precios_alimentos,
//  Fotos_Alimentos): eso no cuenta como "fila ocupada" para el bloque
//  de registro.
//
//  Por qué hace falta: sheet.appendRow() (usado en appendRowByHeader,
//  la función genérica) siempre agrega DESPUÉS de la última fila con
//  contenido en CUALQUIER columna de la hoja completa. Como el
//  catálogo de productos (K en adelante) vive en filas fijas que no
//  se mueven —y desde el fix de handleDeleteAlimentacion, las filas
//  de registro eliminadas quedan vacías en A-J en vez de
//  desaparecer—, appendRow terminaba agregando los registros nuevos
//  siempre al final de TODA la hoja (después de la última fila de
//  catálogo), dejando huecos vacíos sin usar en las filas 2 en
//  adelante que sí tenían A-J libre. Esta función devuelve esa
//  primera fila libre para reutilizarla, en vez de seguir creciendo
//  hacia abajo.
//
//  Devuelve el número de fila (1-indexed, formato Sheets) de la
//  primera fila libre encontrada. Si NINGUNA fila existente tiene A-J
//  vacío (planilla sin huecos, todo ocupado), devuelve
//  sheet.getLastRow() + 1, es decir: agregar una fila nueva al final,
//  el mismo comportamiento que tendría appendRow en ese caso.
// ═══════════════════════════════════════════════════════════════
function findFirstEmptyRowInAlimentacionRegistro(sheet) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return 2; // hoja recién creada, sin filas de datos todavía

  var lastCol = sheet.getLastColumn();
  var colsToCheck = getRegistroWidth(sheet, COLUMNAS_REGISTRO_ALIMENTACION);
  // Solo se leen las columnas A-J (colsToCheck), nunca K en adelante:
  // el catálogo no debe influir en qué cuenta como "fila libre".
  var data = sheet.getRange(2, 1, lastRow - 1, colsToCheck).getDisplayValues();

  for (var i = 0; i < data.length; i++) {
    var row = data[i];
    var rowHasContent = false;
    for (var c = 0; c < row.length; c++) {
      if (String(row[c]).trim() !== '') { rowHasContent = true; break; }
    }
    if (!rowHasContent) {
      return i + 2; // +2: i es 0-indexed sobre data, que arranca en la fila 2 real
    }
  }

  return lastRow + 1; // no había huecos: agregar al final, como appendRow
}

// ═══════════════════════════════════════════════════════════════
//  appendRowByHeaderInFirstEmptySlot — Variante de appendRowByHeader
//  (que NO se toca, sigue usándose tal cual para CuidadoDomicilio,
//  donde este problema no aplica) específica para la hoja
//  "Alimentacion": en vez de sheet.appendRow() —que siempre va al
//  final de toda la hoja—, escribe los valores en la primera fila
//  libre según findFirstEmptyRowInAlimentacionRegistro de arriba,
//  usando setValues sobre el rango A:J de esa fila puntual (nunca
//  toca K en adelante, sea cual sea el contenido que ya tenga esa
//  fila ahí).
//
//  Firma y forma de uso idénticas a appendRowByHeader (mismo
//  headerMap, mismo objeto `values` {headerNormalizado: valor}), para
//  que el cambio en handleAddAlimentacion sea mínimo.
// ═══════════════════════════════════════════════════════════════
function appendRowByHeaderInFirstEmptySlot(sheet, headerMap, values) {
  var targetRow = findFirstEmptyRowInAlimentacionRegistro(sheet);

  var lastCol = sheet.getLastColumn();
  var colsToWrite = getRegistroWidth(sheet, COLUMNAS_REGISTRO_ALIMENTACION);
  var row = new Array(colsToWrite).fill('');
  for (var key in values) {
    if (!values.hasOwnProperty(key)) continue;
    if (headerMap.hasOwnProperty(key) && headerMap[key] < colsToWrite) {
      row[headerMap[key]] = values[key];
    }
  }

  sheet.getRange(targetRow, 1, 1, colsToWrite).setValues([row]);
  return targetRow;
}

// ═══════════════════════════════════════════════════════════════
//  findRowByIdAlimentacion / findRowByIdDesparasitario — FIX v27.
//  Busca, dentro del rango de registro (A..Id) de la hoja
//  correspondiente, si ya existe una fila con ese Id exacto. Devuelve
//  el número de fila (1-indexed) si la encuentra, o -1 si no existe
//  ninguna fila con ese Id todavía.
//
//  Se usan desde handleAddAlimentacion / handleAddDesparasitario ANTES
//  de decidir si escribir en una fila libre nueva o actualizar una ya
//  existente. Esto es lo que corta la duplicación en la fuente: si la
//  acción de alta se dispara dos veces con el mismo id (doble tap,
//  reintento de red, doble render del lado del cliente, etc.), la
//  segunda llamada encuentra la fila que ya escribió la primera y la
//  actualiza en el lugar, en vez de crear una fila hermana con el
//  mismo Id repetido.
//
//  Mismo criterio de columna Id que ya usa updateRowByHeader (busca
//  'id' en headerMap), así que sigue funcionando igual aunque el
//  orden de columnas cambie en el futuro.
// ═══════════════════════════════════════════════════════════════
function findRowByIdAlimentacion(sheet, headerMap, id) {
  if (!id || !headerMap.hasOwnProperty('id')) return -1;
  var colId = headerMap['id'];
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;
  var data = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getDisplayValues();
  for (var i = 0; i < data.length; i++) {
    if (String(data[i][colId] || '').trim() === String(id).trim()) {
      return i + 2; // +2: i es 0-indexed sobre data, que arranca en la fila 2 real
    }
  }
  return -1;
}

// ═══════════════════════════════════════════════════════════════
//  findFirstEmptyRowInDesparasitacionRegistro /
//  appendRowByHeaderInFirstEmptySlotDesparasitacion — Equivalentes
//  exactos de findFirstEmptyRowInAlimentacionRegistro y
//  appendRowByHeaderInFirstEmptySlot, pero para la hoja
//  "Desparasitarios" y su rango de registro A-G
//  (COLUMNAS_REGISTRO_DESPARASITACION), en vez de A-J.
//
//  DESDE v21: handleAddDesparasitario usa
//  appendRowByHeaderInFirstEmptySlotDesparasitacion en vez de
//  appendRowByHeader. Motivo: exactamente el mismo problema que ya se
//  había resuelto para "Alimentacion" — sheet.appendRow() siempre
//  manda la fila nueva al final de TODA la hoja (después del catálogo
//  en H en adelante), en vez de reutilizar los huecos A-G que deja un
//  registro eliminado (handleDeleteDesparasitario usa clearContent
//  desde v20, así que esos huecos existen y hay que reutilizarlos).
//
//  FIX v22: COLUMNAS_REGISTRO_DESPARASITACION bajó de 8 a 7, así que
//  estas dos funciones ahora chequean/escriben solo A-G y dejan H
//  (catálogo) completamente afuera del chequeo de fila libre.
//
//  NOTA v23: esta función SOLO escribe el rango A-G (colsToWrite,
//  derivado de COLUMNAS_REGISTRO_DESPARASITACION/getRegistroWidth).
//  Las 5 columnas de pago que se agregan a partir de v23 quedan fuera
//  de este rango a propósito y se escriben aparte, con setValue
//  puntual, en handleAddDesparasitario — así esta función y su par de
//  Alimentación permanecen sin tocar.
// ═══════════════════════════════════════════════════════════════
function findFirstEmptyRowInDesparasitacionRegistro(sheet) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return 2;

  var lastCol = sheet.getLastColumn();
  var colsToCheck = getRegistroWidth(sheet, COLUMNAS_REGISTRO_DESPARASITACION);
  var data = sheet.getRange(2, 1, lastRow - 1, colsToCheck).getDisplayValues();

  for (var i = 0; i < data.length; i++) {
    var row = data[i];
    var rowHasContent = false;
    for (var c = 0; c < row.length; c++) {
      if (String(row[c]).trim() !== '') { rowHasContent = true; break; }
    }
    if (!rowHasContent) {
      return i + 2;
    }
  }

  return lastRow + 1;
}

function updateRowByHeader(sheet, headerMap, id, values) {
  if (!headerMap.hasOwnProperty('id')) return -1;
  var colId = headerMap['id'];
  var data = sheet.getDataRange().getDisplayValues();
  var rowNumber = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][colId] || '').trim() === id) { rowNumber = i + 1; break; }
  }
  if (rowNumber === -1) return -1;

  for (var key in values) {
    if (!values.hasOwnProperty(key)) continue;
    if (headerMap.hasOwnProperty(key)) {
      sheet.getRange(rowNumber, headerMap[key] + 1).setValue(values[key]);
    }
  }
  return rowNumber;
}

// ═══════════════════════════════════════════════════════════════
//  handleAddAlimentacion — CAMBIO v25: se agregan las 5 columnas de
//  pago (Metodo_Pago, Alias_Pago, Estado_Pago, Referencia_Pago,
//  Numero_Pedido) al array de headers de getOrCreateSheet, y se
//  escriben con un bloque `datosPago` aparte DESPUÉS del append
//  principal, mismo patrón EXACTO que ya usa handleAddDesparasitario
//  desde v23 (ver el comentario grande arriba de esa función).
//
//  Por qué aparte y no dentro de appendRowByHeaderInFirstEmptySlot:
//  esa función recorta en colsToWrite = getRegistroWidth(...), que
//  termina en la columna Id — cualquier header con índice >= Id queda
//  descartado por el guard `headerMap[key] < colsToWrite`. Las 5
//  columnas de pago se agregan DESPUÉS de Id en los headers (mismo
//  criterio que ya se aplicó a Desparasitarios), así que quedan fuera
//  de ese rango a propósito y necesitan su propio setValue por
//  columna, con la misma rowNumber que ya devolvió el append
//  principal.
//
//  Si p.paymentMethod / p.paymentAlias / p.paymentStatus /
//  p.paymentRef / p.orderNumber no vienen (altas que no pasan por el
//  wizard de pago), se guardan como string vacío — mismo criterio que
//  el resto del código usa para cualquier campo opcional ausente.
//
//  CAMBIO v27: antes de decidir en qué fila escribir, se chequea si el
//  `id` que llegó ya existe en la hoja (findRowByIdAlimentacion). Si
//  existe, se ACTUALIZA esa fila en el lugar (mismo camino que
//  handleUpdateAlimentacion) en vez de buscar una fila libre y crear
//  una fila nueva. Esto evita que una alta disparada dos veces con el
//  mismo id (doble tap, reintento de red, doble render) termine
//  creando dos filas distintas con el mismo Id repetido — mismo fix
//  aplicado a handleAddDesparasitario, ver su comentario para el
//  detalle completo del síntoma que esto corrige. Si el id no existía
//  todavía (alta real nueva, el caso normal), el comportamiento es
//  EXACTAMENTE el mismo de antes.
// ═══════════════════════════════════════════════════════════════
// ═══════════════════════════════════════════════════════════════
// FECHAREGISTRO — fecha contable del alta
// Se guarda una vez, al crear el registro, y NO cambia al editar.
// El frontend financiero usa exclusivamente esta columna para asignar
// ingresos a un mes. No se usan Ultima/Proxima/FechaInicio.
// ═══════════════════════════════════════════════════════════════
function ensureHeaderColumn(sheet, headerName) {
  var target = String(headerName || '').trim();
  if (!target) return;
  var map = getHeaderMap(sheet);
  var norm = target.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '').replace(/[\s_]/g, '');
  if (map.hasOwnProperty(norm)) return;
  var lastCol = sheet.getLastColumn();
  sheet.getRange(1, lastCol + 1).setValue(target);
}

function fechaRegistroAhora() {
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd'T'HH:mm:ss");
}

// Ejecutar MANUALMENTE una sola vez si se necesita clasificar registros
// históricos que quedaron sin FechaRegistro. No inventa fechas: recibe una
// fecha explícita (yyyy-MM-dd) y solo completa celdas vacías.
function completarFechaRegistroHistoricos(fechaISO) {
  var fecha = String(fechaISO || '').trim();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(fecha)) throw new Error('Usá una fecha con formato yyyy-MM-dd');
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var specs = [
    ['Alimentacion', ['Id','Nombre','Mail','Marca','precio_registro_de_alimentacion','Porcion','Frecuencia','Ultima','Proxima','Notas']],
    ['Desparasitarios', ['Id','Nombre','Mail','Producto','Descripcion','precio_registro_de_desparasitacion','Ultima','Proxima']],
    ['CuidadoDomicilio', ['Id','Nombre','Mail','Descripcion','FechaInicio','FechaFin','Estado','Costo']]
  ];
  var total = 0;
  specs.forEach(function(spec) {
    var sheet = ss.getSheetByName(spec[0]);
    if (!sheet) return;
    ensureHeaderColumn(sheet, 'FechaRegistro');
    var map = getHeaderMap(sheet);
    if (!map.hasOwnProperty('fecharegistro') || !map.hasOwnProperty('id')) return;
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return;
    var data = sheet.getDataRange().getDisplayValues();
    var dateCol = map['fecharegistro'];
    var idCol = map['id'];
    for (var i=1;i<data.length;i++) {
      if (String(data[i][idCol] || '').trim() && !String(data[i][dateCol] || '').trim()) {
        sheet.getRange(i+1, dateCol+1).setValue(fecha);
        total++;
      }
    }
  });
  return 'Fechas completadas: ' + total;
}

function handleAddAlimentacion(p, eForJSONP) {
  try {
    var nombre = String(p.name || '').trim();
    var marca = String(p.brand || '').trim();
    if (!nombre || !marca) {
      return returnJSONP([{ ok: false, error: 'Faltan mascota o marca' }], eForJSONP);
    }
    var id = String(p.id || '').trim() || Utilities.getUuid();
    var sheet = getOrCreateSheet('Alimentacion',
      ['Id', 'Nombre', 'Mail', 'Marca', 'precio_registro_de_alimentacion', 'Porcion', 'Frecuencia', 'Ultima', 'Proxima', 'Notas',
        'Metodo_Pago', 'Alias_Pago', 'Estado_Pago', 'Referencia_Pago', 'Numero_Pedido', 'FechaRegistro']);
    ensureHeaderColumn(sheet, 'FechaRegistro');
    var headerMap = getHeaderMap(sheet);

    var precioRegistro = resolveCatalogPrecio(marca);

    // FIX v27: si este id YA existe en la hoja (alta repetida con el
    // mismo id), se actualiza esa fila en vez de crear una nueva. Ver
    // findRowByIdAlimentacion y el comentario grande de esta función.
    var existingRow = findRowByIdAlimentacion(sheet, headerMap, id);

    var rowNumber;
    if (existingRow !== -1) {
      rowNumber = updateRowByHeader(sheet, headerMap, id, {
        'nombre': nombre,
        'mail': String(p.mail || '').trim().toLowerCase(),
        'marca': marca,
        'precioregistrodealimentacion': precioRegistro,
        'porcion': String(p.portion || '').trim(),
        'frecuencia': String(p.freq || '').trim(),
        'ultima': String(p.last || '').trim(),
        'proxima': String(p.next || '').trim(),
        'notas': String(p.notes || '').trim(),
        'cliente_id': String(p.clienteId || '').trim()
      });
    } else {
      // appendRowByHeaderInFirstEmptySlot en vez de appendRowByHeader. Ver
      // comentario junto a su definición: escribe en la primera fila con
      // A-J vacío (reutilizando huecos dejados por handleDeleteAlimentacion)
      // en vez de ir siempre al final de toda la hoja, que es lo que hacía
      // que los registros nuevos "saltaran" al final en vez de ocupar la
      // fila 2 en adelante. Solo escribe A..Id (colsToWrite interno); las
      // 5 columnas de pago quedan fuera de ese rango y se escriben abajo.
      rowNumber = appendRowByHeaderInFirstEmptySlot(sheet, headerMap, {
        'id': id,
        'nombre': nombre,
        'mail': String(p.mail || '').trim().toLowerCase(),
        'marca': marca,
        'precioregistrodealimentacion': precioRegistro,
        'porcion': String(p.portion || '').trim(),
        'frecuencia': String(p.freq || '').trim(),
        'ultima': String(p.last || '').trim(),
        'proxima': String(p.next || '').trim(),
        'notas': String(p.notes || '').trim()
      });
    if (existingRow === -1) {
      var fechaCol = getHeaderMap(sheet)['fecharegistro'];
      if (fechaCol !== undefined) sheet.getRange(rowNumber, fechaCol + 1).setValue(fechaRegistroAhora());
    }

    }

    // Formato de moneda ($) en la celda de precio de ESTA fila puntual.
    // Ver aplicarFormatoMonedaPrecioAlimentacion más abajo: el valor
    // sigue siendo un número real (setValue de un number, no un string
    // con "$"), solo cambia cómo se VE en pantalla.
    aplicarFormatoMonedaPrecioAlimentacion(sheet, headerMap, rowNumber);

    // Datos de pago (v25): se escriben aparte porque
    // appendRowByHeaderInFirstEmptySlot recorta en Id y nunca llega a
    // estas columnas. Se guarda '' si el campo no vino (altas que no
    // pasan por el wizard de pago nunca mandan estos 5 parámetros).
    // Mismo bloque exacto que ya usa handleAddDesparasitario. Se aplica
    // también en el camino de "ya existía" (existingRow !== -1) porque
    // updateRowByHeader no toca estas 5 columnas por su cuenta.
    var datosPago = {
      'metodopago': String(p.paymentMethod || '').trim(),
      'aliaspago': String(p.paymentAlias || '').trim(),
      'estadopago': String(p.paymentStatus || '').trim(),
      'referenciapago': String(p.paymentRef || '').trim(),
      'numeropedido': String(p.orderNumber || '').trim()
    };
    for (var key in datosPago) {
      if (!datosPago.hasOwnProperty(key)) continue;
      if (headerMap.hasOwnProperty(key)) {
        sheet.getRange(rowNumber, headerMap[key] + 1).setValue(datosPago[key]);
      }
    }

    return returnJSONP([{ ok: true, id: id, precio: precioRegistro }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

function handleUpdateAlimentacion(p, eForJSONP) {
  try {
    var id = String(p.id || '').trim();
    var nombre = String(p.name || '').trim();
    var marca = String(p.brand || '').trim();
    if (!id) {
      return returnJSONP([{ ok: false, error: 'Falta el id del registro a editar' }], eForJSONP);
    }
    if (!nombre || !marca) {
      return returnJSONP([{ ok: false, error: 'Faltan mascota o marca' }], eForJSONP);
    }
    var sheet = getOrCreateSheet('Alimentacion',
      ['Id', 'Nombre', 'Mail', 'Marca', 'precio_registro_de_alimentacion', 'Porcion', 'Frecuencia', 'Ultima', 'Proxima', 'Notas',
        'Metodo_Pago', 'Alias_Pago', 'Estado_Pago', 'Referencia_Pago', 'Numero_Pedido', 'FechaRegistro']);
    var headerMap = getHeaderMap(sheet);

    // El precio_registro_de_alimentacion es un SNAPSHOT histórico.
    // Cambiar precios_alimentos NO debe modificar compras ya registradas.
    // Solo se obtiene un precio nuevo si el registro cambia realmente de
    // producto/marca; editar porción, frecuencia o fechas conserva el
    // precio original guardado en la fila.
    var existingPrice = 0;
    var existingMarca = '';
    var idCol = headerMap.hasOwnProperty('id') ? headerMap['id'] : -1;
    var priceCol = headerMap.hasOwnProperty('precioregistrodealimentacion') ? headerMap['precioregistrodealimentacion'] : -1;
    var marcaCol = headerMap.hasOwnProperty('marca') ? headerMap['marca'] : -1;
    if (idCol !== -1 && priceCol !== -1 && marcaCol !== -1) {
      var existingData = sheet.getDataRange().getDisplayValues();
      for (var ep = 1; ep < existingData.length; ep++) {
        if (String(existingData[ep][idCol] || '').trim() === id) {
          existingPrice = parsePrecioCLP(existingData[ep][priceCol]);
          existingMarca = String(existingData[ep][marcaCol] || '').trim();
          break;
        }
      }
    }
    var productChanged = normalizeNombreProducto(existingMarca) !== normalizeNombreProducto(marca);
    var precioRegistro = productChanged ? resolveCatalogPrecio(marca) : existingPrice;

    var rowNumber = updateRowByHeader(sheet, headerMap, id, {
      'nombre': nombre,
      'mail': String(p.mail || '').trim().toLowerCase(),
      'marca': marca,
      'precioregistrodealimentacion': precioRegistro,
      'porcion': String(p.portion || '').trim(),
      'frecuencia': String(p.freq || '').trim(),
      'ultima': String(p.last || '').trim(),
      'proxima': String(p.next || '').trim(),
      'notas': String(p.notes || '').trim(),
      'cliente_id': String(p.clienteId || '').trim()
    });
    if (rowNumber === -1) {
      return returnJSONP([{ ok: false, error: 'No se encontró el registro con ese Id en la planilla (correr migrarIdAlimentacion si la hoja es vieja)' }], eForJSONP);
    }

    // Mismo formato de moneda que handleAddAlimentacion. Necesario acá
    // también porque updateRowByHeader usa setValue (no toca el formato
    // de número de la celda), así que si la fila editada nunca había
    // pasado por el formateo (por ejemplo, un registro viejo migrado
    // antes de v17), sin esta línea quedaría mostrando el número plano
    // sin "$" aunque el valor ya esté recalculado correctamente.
    aplicarFormatoMonedaPrecioAlimentacion(sheet, headerMap, rowNumber);

    // NOTA v25: igual que handleUpdateDesparasitario, esta función NO
    // toca las columnas de pago. El index no manda paymentMethod/
    // paymentAlias/paymentStatus/paymentRef/orderNumber en el update de
    // Alimentación (editar un registro solo permite tocar mascota/
    // marca/porción/fechas, nunca el dato de pago ya asentado). Si el
    // frontend empieza a mandar esos campos también en el update, hay
    // que sumar el mismo bloque `datosPago` que ya usa
    // handleAddAlimentacion acá también.

    return returnJSONP([{ ok: true, id: id, precio: precioRegistro }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

// ═══════════════════════════════════════════════════════════════
//  handleDeleteAlimentacion — en vez de deleteRow, se limpia SOLO el
//  rango de columnas A hasta COLUMNAS_REGISTRO_ALIMENTACION (hoy = 10,
//  o sea A-J) de esa fila puntual, con clearContent(). Las columnas K
//  en adelante (catálogo) no se tocan en absoluto: ni se leen, ni se
//  escriben, ni se corren de lugar. Esto evita el bug de deleteRow que
//  arrasaba con las columnas de catálogo (K-N) al borrar un registro
//  de compra, porque catálogo y registro viven en las mismas filas
//  físicas de la hoja.
//
//  Efecto observable: al eliminar un registro de alimentación, esa
//  fila deja de tener Id/Nombre/Mail/Marca/precio/Porcion/Frecuencia/
//  Ultima/Proxima/Notas (columnas A-J, todas vacías), pero si esa
//  misma fila tenía además un producto de catálogo en K/L/M/N, ese
//  producto de catálogo permanece exactamente igual que antes de
//  eliminar. La fila NO se elimina de la hoja; simplemente su bloque
//  de registro queda vacío. Esto no genera registros fantasma en el
//  frontend porque handleGetAlimentacion (vía sheetToObjects) y
//  handleGetCatalogoAlimentos ya descartan/ignoran independientemente
//  cualquier fila sin contenido en su respectivo bloque de columnas.
//
//  NOTA v25: las 5 columnas de pago agregadas al final de los headers
//  quedan DELIBERADAMENTE fuera de colsToClear (que sigue siendo
//  getRegistroWidth(sheet, COLUMNAS_REGISTRO_ALIMENTACION), o sea
//  A-J): al borrar un registro, el dato de con qué método/estado se
//  pagó permanece en la hoja aunque el registro de compra en sí se
//  limpie. Mismo criterio exacto que ya aplica
//  handleDeleteDesparasitario a sus propias columnas de pago desde
//  v23.
// ═══════════════════════════════════════════════════════════════
function handleDeleteAlimentacion(p, eForJSONP) {
  try {
    var id = String(p.id || '').trim();
    if (!id) {
      return returnJSONP([{ ok: false, error: 'Falta el id del registro a eliminar' }], eForJSONP);
    }
    var sheet = getOrCreateSheet('Alimentacion',
      ['Id', 'Nombre', 'Mail', 'Marca', 'precio_registro_de_alimentacion', 'Porcion', 'Frecuencia', 'Ultima', 'Proxima', 'Notas',
        'Metodo_Pago', 'Alias_Pago', 'Estado_Pago', 'Referencia_Pago', 'Numero_Pedido', 'FechaRegistro']);
    var data = sheet.getDataRange().getDisplayValues();
    if (data.length < 2) {
      return returnJSONP([{ ok: false, error: 'La hoja está vacía' }], eForJSONP);
    }
    var headers = data[0];
    var colId = -1;
    for (var h = 0; h < headers.length; h++) {
      if (String(headers[h]).trim().toLowerCase() === 'id') { colId = h; break; }
    }
    if (colId === -1) {
      return returnJSONP([{ ok: false, error: 'La hoja no tiene columna Id (registros viejos, sin id, no se pueden borrar así)' }], eForJSONP);
    }

    // Cuántas columnas tiene HOY la hoja en total (para no pasarnos si
    // por algún motivo la hoja tuviera menos columnas que
    // COLUMNAS_REGISTRO_ALIMENTACION, caso borde de una hoja recién
    // creada con menos headers).
    var lastCol = sheet.getLastColumn();
    var colsToClear = getRegistroWidth(sheet, COLUMNAS_REGISTRO_ALIMENTACION);

    var deletedRows = 0;
    // Recorremos de arriba hacia abajo: a diferencia de deleteRow (que
    // corre las filas de abajo hacia arriba y por eso había que iterar
    // al revés para no desalinear índices), clearContent NO mueve
    // ninguna fila de lugar, así que no hay riesgo de desalineación y
    // el orden de recorrido es indistinto.
    for (var i = 1; i < data.length; i++) {
      var rowId = String(data[i][colId] || '').trim();
      if (rowId === id) {
        // i es 0-indexed sobre `data`; la fila real de Sheets es i + 1.
        // clearContent SOLO sobre columnas 1..colsToClear (A hasta J):
        // el catálogo de K en adelante (y las columnas de pago
        // agregadas en v25) no se tocan.
        sheet.getRange(i + 1, 1, 1, colsToClear).clearContent();
        deletedRows++;
      }
    }
    return returnJSONP([{ ok: deletedRows > 0, deleted: deletedRows }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

// ═══════════════════════════════════════════════════════════════
//  handleAddDesparasitario — Resuelve precioRegistro contra el
//  catálogo ANTES de armar el objeto de columnas (mismo patrón exacto
//  que ya usa handleAddAlimentacion con resolveCatalogPrecio para la
//  hoja Alimentacion), lo agrega como
//  'precioregistrodedesparasitacion' al objeto de valores, aplica
//  formato moneda a esa celda con
//  aplicarFormatoMonedaPrecioDesparasitacion, y devuelve { ok, id,
//  precio } en la respuesta JSON.
//
//  Usa appendRowByHeaderInFirstEmptySlotDesparasitacion en vez de
//  appendRowByHeader. Antes, appendRowByHeader hacía
//  sheet.appendRow(), que siempre agrega la fila nueva DESPUÉS de la
//  última fila con contenido en CUALQUIER columna de TODA la hoja.
//  Como el catálogo (Nombre_desparasitarios_segun_kink, columna H)
//  ocupa filas fijas, cada alta nueva caía siempre después de la
//  última fila de catálogo, dejando vacías las filas de arriba en
//  vez de ocuparlas. Ahora, con
//  appendRowByHeaderInFirstEmptySlotDesparasitacion, el alta busca
//  primero la primera fila con A-G completamente vacío y escribe ahí
//  (reutilizando también los huecos que deja
//  handleDeleteDesparasitario desde su fix de clearContent en v20); si
//  no hay ninguna fila libre, agrega al final, igual que appendRow. El
//  catálogo en H sigue sin tocarse en absoluto en ningún caso.
//
//  El campo 'desc' que puede seguir llegando desde un frontend viejo
//  en caché ya no se usa para nada especial: si no llega, se guarda el
//  mismo valor que 'producto' en la columna Descripcion (para no dejar
//  esa columna vacía ni tener que borrarla de la hoja).
//
//  FIX v23: el flujo de pago (dwCompleteOrderAfterPayment en
//  index.html) manda 5 parámetros más que antes se ignoraban por
//  completo porque no estaban ni en los headers de getOrCreateSheet ni
//  en el objeto de `values` de más abajo: paymentMethod, paymentAlias,
//  paymentStatus, paymentRef, orderNumber. Se agregan como columnas
//  nuevas al final de los headers (Metodo_Pago, Alias_Pago,
//  Estado_Pago, Referencia_Pago, Numero_Pedido) y se escriben con
//  appendRowByHeader normal en headerMap, PERO por fuera del rango que
//  escribe appendRowByHeaderInFirstEmptySlotDesparasitacion (que corta
//  en colsToWrite = getRegistroWidth(...) = A..Id, o sea nunca llega a
//  estas 5 columnas nuevas): por eso se hace un setValues aparte,
//  puntual, en la misma rowNumber que ya devolvió el append de A-G.
//  Si algún alta llega sin esos 5 campos (el modal simple de
//  saveHealth() y la copia de duplicateDeworm() nunca los mandan), se
//  guardan como string vacío, igual que hace el resto del código con
//  cualquier campo opcional ausente.
//
//  FIX v27 (duplicación de registros en Cobros & Costos): antes, esta
//  función SIEMPRE creaba una fila nueva, sin importar si el `id` que
//  llegaba ya existía en la hoja. Si la acción addDesparasitario se
//  disparaba dos veces con el mismo id (doble tap del botón, un
//  timeout de fetchJSONP que hace que el cliente dé la llamada por
//  perdida cuando en realidad SÍ había llegado al servidor, un doble
//  render del lado del cliente, etc.), el resultado eran DOS filas
//  distintas en la hoja con el mismo Id repetido. La sincronización
//  (getServiciosCliente) devuelve ambas filas tal cual están, y el
//  frontend las mapea 1 a 1 a state.health sin deduplicar por id —
//  terminaban como dos registros distintos en memoria para un único
//  evento real, duplicando tanto el conteo de "Registros Activos"
//  como los ingresos de Desparasitarios en Cobros & Costos.
//
//  Ahora, antes de decidir dónde escribir, se busca si el `id` ya
//  existe (findRowByIdDesparasitario). Si existe, se actualiza esa
//  fila en el lugar (mismo camino que handleUpdateDesparasitario) en
//  vez de buscar una fila libre y crear una fila nueva — así una
//  segunda llamada con el mismo id nunca puede duplicar, sea cual sea
//  la causa de que se haya disparado dos veces. Si no existe (caso
//  normal, alta real nueva), el comportamiento es EXACTAMENTE el
//  mismo de antes: appendRowByHeaderInFirstEmptySlotDesparasitacion
//  busca la primera fila libre y escribe ahí.
// ═══════════════════════════════════════════════════════════════
function handleAddDesparasitario(p, eForJSONP) {
  try {
    var nombre = String(p.name || '').trim();
    var producto = String(p.product || '').trim();

    // Si no llega 'desc' (frontend actualizado ya no lo manda), se usa
    // el propio nombre del producto para la columna Descripcion, en vez
    // de dejarla vacía.
    var descripcion = String(p.desc || '').trim() || producto;

    // FIX v27: si este id YA existe en la hoja (alta repetida con el
    // mismo id), se actualiza esa fila en vez de crear una nueva. Ver
    // findRowByIdDesparasitario y el comentario grande de esta función

    var existingRow = findRowByIdDesparasitario(sheet, headerMap, id);

      // en la primera fila con A-G vacío (reutilizando huecos dejados
      // por handleDeleteDesparasitario) en vez de ir siempre al final de
      // toda la hoja, que era lo que hacía que los registros nuevos
      // "saltaran" a la fila 6 en vez de ocupar la fila 2 en adelante.
      // Esta llamada solo escribe A..Id (colsToWrite interno); las 5
      // columnas de pago quedan fuera de ese rango y se escriben abajo.
      rowNumber = appendRowByHeaderInFirstEmptySlotDesparasitacion(sheet, headerMap, {
        'id': id,
        'nombre': nombre,
        'mail': String(p.mail || '').trim().toLowerCase(),
        'producto': producto,
        'descripcion': descripcion,
        'precioregistrodedesparasitacion': precioRegistro,
        'ultima': String(p.last || '').trim(),
        'proxima': String(p.next || '').trim(),
        'cliente_id': String(p.clienteId || '').trim()
      });
    }

    // Formato de moneda ($) en la celda de precio de ESTA fila puntual,
    // mismo patrón exacto que aplicarFormatoMonedaPrecioAlimentacion.
    aplicarFormatoMonedaPrecioDesparasitacion(sheet, headerMap, rowNumber);

    // Datos de pago (v23): se escriben aparte porque
    // appendRowByHeaderInFirstEmptySlotDesparasitacion recorta en Id y
    // nunca llega a estas columnas. Se guarda '' si el campo no vino
    // (altas que no pasan por el wizard de pago, ej. saveHealth() o
    // duplicateDeworm(), nunca mandan estos 5 parámetros). Se aplica
    // también en el camino de "ya existía" (existingRow !== -1) porque
    // updateRowByHeader no toca estas 5 columnas por su cuenta.
    var datosPago = {
      'metodopago': String(p.paymentMethod || '').trim(),
      'aliaspago': String(p.paymentAlias || '').trim(),
      'estadopago': String(p.paymentStatus || '').trim(),
      'referenciapago': String(p.paymentRef || '').trim(),
      'numeropedido': String(p.orderNumber || '').trim()
    };
    for (var key in datosPago) {
      if (!datosPago.hasOwnProperty(key)) continue;
      if (headerMap.hasOwnProperty(key)) {
        sheet.getRange(rowNumber, headerMap[key] + 1).setValue(datosPago[key]);
      }
    }

    return returnJSONP([{ ok: true, id: id, precio: precioRegistro }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

// ═══════════════════════════════════════════════════════════════
//  handleUpdateDesparasitario — mismo criterio que
//  handleAddDesparasitario de arriba: al editar un registro, si
//  cambió el producto elegido, el precio_registro_de_desparasitacion
//  se vuelve a resolver contra el catálogo ACTUAL (mismo patrón que
//  handleUpdateAlimentacion). Sigue usando updateRowByHeader, que
//  busca la fila por Id (no por posición), así que no le afecta el
//  fix de dónde cae el alta.
//
//  NOTA v23: esta función NO se modifica. El index nunca manda
//  paymentMethod/paymentAlias/paymentStatus/paymentRef/orderNumber en
//  la URL de updateDesparasitario (ver saveDeworm en index.html, línea
//  ~9593): editar un registro solo permite tocar mascota/producto/
//  fechas, nunca el dato de pago ya asentado. Si el día de mañana el
//  frontend empieza a mandar esos campos también en el update, hay que
//  sumar el mismo bloque `datosPago` que ya usa handleAddDesparasitario
//  acá también.
// ═══════════════════════════════════════════════════════════════
function handleUpdateDesparasitario(p, eForJSONP) {
  try {
    var id = String(p.id || '').trim();
    var nombre = String(p.name || '').trim();
    var producto = String(p.product || '').trim();
    if (!id) {
      return returnJSONP([{ ok: false, error: 'Falta el id del registro a editar' }], eForJSONP);
    }
    if (!nombre || !producto) {
      return returnJSONP([{ ok: false, error: 'Faltan mascota o producto' }], eForJSONP);
    }
    var sheet = getOrCreateSheet('Desparasitarios',
      ['Id', 'Nombre', 'Mail', 'Producto', 'Descripcion', 'precio_registro_de_desparasitacion', 'Ultima', 'Proxima',
        'Metodo_Pago', 'Alias_Pago', 'Estado_Pago', 'Referencia_Pago', 'Numero_Pedido', 'FechaRegistro']);
    var headerMap = getHeaderMap(sheet);

    // El precio_registro_de_desparasitacion es un SNAPSHOT histórico.
    // Cambiar precios_desparasitarios NO debe modificar aplicaciones ya
    // registradas. Solo se toma el precio vigente del catálogo cuando el
    // registro cambia realmente de producto.
    var existingPrice = 0;
    var existingProduct = '';
    var idCol = headerMap.hasOwnProperty('id') ? headerMap['id'] : -1;
    var priceCol = headerMap.hasOwnProperty('precioregistrodedesparasitacion') ? headerMap['precioregistrodedesparasitacion'] : -1;
    var productCol = headerMap.hasOwnProperty('producto') ? headerMap['producto'] : -1;
    if (idCol !== -1 && priceCol !== -1 && productCol !== -1) {
      var existingData = sheet.getDataRange().getDisplayValues();
      for (var ep = 1; ep < existingData.length; ep++) {
        if (String(existingData[ep][idCol] || '').trim() === id) {
          existingPrice = parsePrecioCLP(existingData[ep][priceCol]);
          existingProduct = String(existingData[ep][productCol] || '').trim();
          break;
        }
      }
    }
    var productChanged = normalizeNombreProducto(existingProduct) !== normalizeNombreProducto(producto);
    var precioRegistro = productChanged ? resolveCatalogPrecioDesparasitario(producto) : existingPrice;
    var descripcion = String(p.desc || '').trim() || producto;

    var rowNumber = updateRowByHeader(sheet, headerMap, id, {
      'nombre': nombre,
      'mail': String(p.mail || '').trim().toLowerCase(),
      'producto': producto,
      'descripcion': descripcion,
      'precioregistrodedesparasitacion': precioRegistro,
      'ultima': String(p.last || '').trim(),
      'proxima': String(p.next || '').trim(),
      'cliente_id': String(p.clienteId || '').trim()
    });
    if (rowNumber === -1) {
      return returnJSONP([{ ok: false, error: 'No se encontró el registro con ese Id en la planilla (correr migrarIdDesparasitarios si la hoja es vieja)' }], eForJSONP);
    }

    // Mismo formato de moneda que handleAddDesparasitario. Necesario acá
    // también porque updateRowByHeader usa setValue y no toca el formato
    // de número de la celda: si la fila editada nunca había pasado por
    // el formateo, sin esta línea quedaría mostrando el número plano
    // sin "$" aunque el valor ya esté recalculado correctamente.
    aplicarFormatoMonedaPrecioDesparasitacion(sheet, headerMap, rowNumber);

    return returnJSONP([{ ok: true, id: id, precio: precioRegistro }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

// ═══════════════════════════════════════════════════════════════
//  handleDeleteDesparasitario — se limpia SOLO el rango de columnas A
//  hasta COLUMNAS_REGISTRO_DESPARASITACION (ahora = 7, o sea A-G:
//  Nombre, Mail, Producto, precio_registro_de_desparasitacion,
//  Ultima, Proxima, Id) de esa fila puntual, con clearContent(). La
//  columna H (catálogo) no se toca en absoluto: ni se lee, ni se
//  escribe, ni se corre de lugar. Estos huecos A-G que deja son
//  justamente los que handleAddDesparasitario reutiliza.
//
//  NOTA v23: las 5 columnas de pago agregadas al final de los headers
//  quedan DELIBERADAMENTE fuera de colsToClear (que sigue siendo
//  getRegistroWidth(sheet, COLUMNAS_REGISTRO_DESPARASITACION), o sea
//  A-G): al borrar un registro, el dato de con qué método/estado se
//  pagó permanece en la hoja aunque el registro de compra en sí se
//  limpie. Es el mismo criterio que ya aplica esta función al
//  catálogo en H: solo se toca el bloque de identidad+producto+fechas,
//  nunca lo que esté más a la derecha. Si en algún momento se decide
//  que el dato de pago también debe borrarse junto con el registro,
//  hay que sumar esas 5 columnas al clearContent de abajo a mano (no
//  alcanza con tocar la constante, porque eso afectaría también el
//  chequeo de "fila libre" que usa la misma constante para reutilizar
//  huecos).
// ═══════════════════════════════════════════════════════════════
function handleDeleteDesparasitario(p, eForJSONP) {
  try {
    var id = String(p.id || '').trim();
    if (!id) {
      return returnJSONP([{ ok: false, error: 'Falta el id del registro a eliminar' }], eForJSONP);
    }
    var sheet = getOrCreateSheet('Desparasitarios',
      ['Id', 'Nombre', 'Mail', 'Producto', 'Descripcion', 'precio_registro_de_desparasitacion', 'Ultima', 'Proxima',
        'Metodo_Pago', 'Alias_Pago', 'Estado_Pago', 'Referencia_Pago', 'Numero_Pedido', 'FechaRegistro']);
    var data = sheet.getDataRange().getDisplayValues();
    if (data.length < 2) {
      return returnJSONP([{ ok: false, error: 'La hoja está vacía' }], eForJSONP);
    }
    var headers = data[0];
    var colId = -1;
    for (var h = 0; h < headers.length; h++) {
      if (String(headers[h]).trim().toLowerCase() === 'id') { colId = h; break; }
    }
    if (colId === -1) {
      return returnJSONP([{ ok: false, error: 'La hoja no tiene columna Id todavía (correr migrarIdDesparasitarios una vez desde el editor)' }], eForJSONP);
    }

    // Cuántas columnas tiene HOY la hoja en total (para no pasarnos si
    // por algún motivo la hoja tuviera menos columnas que
    // COLUMNAS_REGISTRO_DESPARASITACION, caso borde de una hoja recién
    // creada con menos headers).
    var lastCol = sheet.getLastColumn();
    var colsToClear = getRegistroWidth(sheet, COLUMNAS_REGISTRO_DESPARASITACION);

    var deletedRows = 0;
    // Recorremos de arriba hacia abajo: a diferencia de deleteRow (que
    // corre las filas de abajo hacia arriba y por eso había que iterar
    // al revés para no desalinear índices), clearContent NO mueve
    // ninguna fila de lugar, así que no hay riesgo de desalineación y
    // el orden de recorrido es indistinto.
    for (var i = 1; i < data.length; i++) {
      var rowId = String(data[i][colId] || '').trim();
      if (rowId === id) {
        // i es 0-indexed sobre `data`; la fila real de Sheets es i + 1.
        // clearContent SOLO sobre columnas 1..colsToClear (A hasta G):
        // el catálogo de H (y las columnas de pago agregadas en v23)
        // no se tocan.
        sheet.getRange(i + 1, 1, 1, colsToClear).clearContent();
        deletedRows++;
      }
    }
    return returnJSONP([{ ok: deletedRows > 0, deleted: deletedRows }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

function normalizeEstadoCuidado(raw) {
  var s = String(raw || '').trim().toLowerCase();
  if (s === 'pending' || s === 'pendiente') return 'Pendiente';
  if (s === 'confirmed' || s === 'confirmado') return 'Confirmado';
  return String(raw || '').trim();
}

function pintarCeldaEstadoCuidado(sheet, headerMap, rowNumber, estadoTraducido) {
  if (!headerMap.hasOwnProperty('estado')) return;
  var colEstado1indexed = headerMap['estado'] + 1;
  var celda = sheet.getRange(rowNumber, colEstado1indexed);
  if (estadoTraducido === 'Confirmado') {
    celda.setBackground('#93C47D');
  } else if (estadoTraducido === 'Pendiente') {
    celda.setBackground('#F6B26B');
  }
}

// ═══════════════════════════════════════════════════════════════
//  aplicarFormatoMonedaPrecioAlimentacion — Aplica formato de número
//  tipo moneda ('$'#,##0) a la celda de la columna
//  precio_registro_de_alimentacion de una fila puntual, para que se
//  VEA con el signo "$" (ej. "$ 74.300"), tal como se ve la columna
//  "precios_alimentos" del catálogo.
//
//  IMPORTANTE: esto es solo FORMATO DE CELDA (setNumberFormat), no
//  cambia el VALOR que hay adentro. La celda sigue conteniendo el
//  número real (ej. 74300, un number de JS), no el string "$74300".
//  Elegir esta vía en vez de guardar un string con "$" es
//  intencional: así el número sigue siendo sumable/filtrable/
//  ordenable en la planilla como cualquier otra columna numérica, y
//  parsePrecioCLP (que ya lee celdas de este mismo formato en el
//  catálogo) sigue funcionando exactamente igual si en algún momento
//  hiciera falta releer este valor.
//
//  Si headerMap no tiene la columna precio_registro_de_alimentacion
//  (hoja vieja sin esa columna todavía), no hace nada.
// ═══════════════════════════════════════════════════════════════
function aplicarFormatoMonedaPrecioAlimentacion(sheet, headerMap, rowNumber) {
  if (!headerMap.hasOwnProperty('precioregistrodealimentacion')) return;
  var colPrecio1indexed = headerMap['precioregistrodealimentacion'] + 1;
  sheet.getRange(rowNumber, colPrecio1indexed).setNumberFormat("\"$\"#,##0");
}

// ═══════════════════════════════════════════════════════════════
//  dedupeByColumn — FIX v27, red de seguridad adicional al fix de
//  handleAddAlimentacion/handleAddDesparasitario (findRowByIdAlimentacion/
//  findRowByIdDesparasitario, que cortan la duplicación EN EL MOMENTO
//  de escribir). Esta función deduplica del lado de LECTURA: si por
//  cualquier motivo la hoja YA tiene filas con el mismo valor en la
//  columna indicada (típicamente 'Id') -- por ejemplo, filas
//  duplicadas que quedaron cargadas ANTES de este fix, o cualquier
//  otra vía de escritura que no pase por los handlers de arriba --,
//  se conserva solo la ÚLTIMA fila con ese valor (la más reciente en
//  la hoja) y se descartan las anteriores. Filas sin valor en esa
//  columna (columna vacía, o headerName que no existe todavía en la
//  hoja) NO se deduplican entre sí: se conservan todas tal cual,
//  porque un valor vacío repetido no significa "mismo registro".
//
//  Se aplica en handleGetServiciosCliente (la única lectura que usa
//  el frontend para sincronizar Alimentación/Desparasitarios/Cuidado
//  a Domicilio, ver loadHealthFoodHomeCareFromSheet en index.html) por
//  columna 'Id', que es el campo que el frontend usa como identidad
//  única de cada registro en state.health/state.food/state.homeCare.
// ═══════════════════════════════════════════════════════════════
function dedupeByColumn(rows, headerName) {
  if (!rows || !rows.length) return rows;
  var seenIndexByKey = {};
  var order = [];
  for (var i = 0; i < rows.length; i++) {
    var raw = rows[i][headerName];
    var key = String(raw == null ? '' : raw).trim();
    if (!key) {
      // Sin valor en la columna: se conserva tal cual, sin deduplicar
      // contra otras filas vacías.
      order.push(rows[i]);
      continue;
    }
    if (seenIndexByKey.hasOwnProperty(key)) {
      // Ya había una fila con este mismo Id más arriba en la hoja: se
      // reemplaza por esta (la más reciente/de más abajo gana), en vez
      // de acumular las dos.
      order[seenIndexByKey[key]] = rows[i];
    } else {
      seenIndexByKey[key] = order.length;
      order.push(rows[i]);
    }
  }
  return order;
}

// ═══════════════════════════════════════════════════════════════
//  resolveCatalogPrecio — busca `nombreProducto` en el catálogo de
//  productos de Alimentos (columnas Nombre_alimentos_segun_link /
//  precios_alimentos de la hoja "Alimentacion") y devuelve su precio
//  actual como número limpio (parsePrecioCLP), o 0 si no lo encuentra
//  o si no hay precio cargado.
//
//  La comparación de nombres pasa por normalizeNombreProducto en
//  ambos lados (nombre buscado y nombre de cada fila del catálogo)
//  antes de comparar con ===, en vez de comparar los strings tal cual
//  llegan, para tolerar diferencias de mayúsculas/espacios entre lo
//  que manda el picker del frontend y lo tipeado en la planilla.
// ═══════════════════════════════════════════════════════════════
function resolveCatalogPrecio(nombreProducto) {

  if (!nombreBuscado) return 0;

  var sheet = getOrCreateSheet('Alimentacion',
    ['Id', 'Nombre', 'Mail', 'Marca', 'precio_registro_de_alimentacion', 'Porcion', 'Frecuencia', 'Ultima', 'Proxima', 'Notas',
      'Metodo_Pago', 'Alias_Pago', 'Estado_Pago', 'Referencia_Pago', 'Numero_Pedido', 'FechaRegistro']);
  var headerMap = getHeaderMap(sheet);

  var colNombreKey = null;
  var nombreCandidates = [
    'nombrealimentossegunlink', 'nombresalimentossegunlink',
    'nombrealimento', 'nombrealimentos', 'nombreproducto', 'producto'
  ];
  for (var nc = 0; nc < nombreCandidates.length; nc++) {
    if (headerMap.hasOwnProperty(nombreCandidates[nc])) { colNombreKey = nombreCandidates[nc]; break; }
  }
  if (!colNombreKey) return 0;

  var colPrecioKey = null;
  var precioCandidates = [
    'preciosalimentos', 'precioalimentos', 'precioalimento',
    'preciosalimento', 'preciosproducto', 'precioproducto', 'precio'
  ];
  for (var pc = 0; pc < precioCandidates.length; pc++) {
    if (headerMap.hasOwnProperty(precioCandidates[pc])) { colPrecioKey = precioCandidates[pc]; break; }
  }
  if (!colPrecioKey) return 0;

  var colNombre = headerMap[colNombreKey];
  var colPrecio = headerMap[colPrecioKey];
  var data = sheet.getDataRange().getDisplayValues();

  for (var i = 1; i < data.length; i++) {
    var nombreFila = normalizeNombreProducto(data[i][colNombre]);
    if (nombreFila === nombreBuscado) {
      return parsePrecioCLP(data[i][colPrecio]);
    }
  }
  return 0;
}

function sheetToObjectsWithRequiredField(sheet, requiredHeader) {
  var data = sheet.getDataRange().getDisplayValues();
  if (data.length < 2) return [];

  var headers = data[0];
  var requiredIndex = -1;
  var wanted = String(requiredHeader || '').trim().toLowerCase();

  for (var h = 0; h < headers.length; h++) {
    if (String(headers[h] || '').trim().toLowerCase() === wanted) {
      requiredIndex = h;
      break;
    }
  }

  // Si por alguna razón la hoja no tiene la columna requerida, no inventamos
  // registros. Devolvemos vacío para endpoints de registro.
  if (requiredIndex === -1) return [];

  var out = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (String(row[requiredIndex] || '').trim() === '') continue;

    var hasContent = false;
    for (var c = 0; c < row.length; c++) {
      if (String(row[c]).trim() !== '') { hasContent = true; break; }
    }
    if (!hasContent) continue;

    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j] !== undefined ? row[j] : '';
    }
    out.push(obj);
  }
  return out;
}

function handleUpdatePeso(mailParam, nameParam, pesoParam, eForJSONP) {
  try {
    var mailBuscado = String(mailParam || '').toLowerCase().trim();
    var nameBuscado = String(nameParam || '').toLowerCase().trim();
    var nuevoPeso = String(pesoParam || '').trim();

    if (!nameBuscado || !nuevoPeso) {
      return returnJSONP([{ ok: false, error: 'Parámetros incompletos' }], eForJSONP);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Mascota Paseo") || ss.getSheets()[0];
    var data = sheet.getDataRange().getDisplayValues();

    var headerRowIndex = -1;
    for (var r = 0; r < Math.min(data.length, 10); r++) {
      for (var c = 0; c < data[r].length; c++) {
        var cell = String(data[r][c]).toLowerCase().trim();
        if (cell.indexOf('nombre') !== -1 || cell.indexOf('mascota') !== -1) {
          headerRowIndex = r; break;
        }
      }
      if (headerRowIndex !== -1) break;
    }

    if (headerRowIndex === -1) {
      return returnJSONP([{ ok: false, error: 'Sin encabezados' }], eForJSONP);
    }

    var headers = data[headerRowIndex];
    var colNombre = -1, colMail = -1, colPeso = -1, colDueno = -1;

    for (var j = 0; j < headers.length; j++) {
      var h = String(headers[j]).toLowerCase().normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '').replace(/[\s_]/g, '').trim();
      if ((h === 'nombre' || h === 'nombremascota' || h === 'mascota') && colNombre === -1) colNombre = j;
      if (h === 'mail' || h === 'email' || h === 'correo') colMail = j;
      if (h === 'pesoactual' || h === 'peso' || h === 'weight') colPeso = j;
      if (h === 'dueno' || h === 'owner' || h === 'propietario') colDueno = j;
    }

    if (colPeso === -1) {
      colPeso = headers.length;
      sheet.getRange(headerRowIndex + 1, colPeso + 1).setValue('Peso Actual');
    }

    var updatedRows = 0;
    for (var i = headerRowIndex + 1; i < data.length; i++) {
      var row = data[i];
      var rowNom = (colNombre !== -1) ? String(row[colNombre] || '').toLowerCase().trim() : '';
      var rowMail = (colMail !== -1) ? String(row[colMail] || '').toLowerCase().trim() : '';
      var rowDueno = (colDueno !== -1) ? String(row[colDueno] || '').toLowerCase().trim() : '';

      if (!rowMail) {
        for (var ci = 0; ci < row.length; ci++) {
          var v = String(row[ci] || '').trim();
          if (v.indexOf('@') !== -1 && v.indexOf('.') !== -1) { rowMail = v.toLowerCase(); break; }
        }
      }

      var match = false;

      if (mailBuscado && rowMail) {
        match = (rowNom === nameBuscado && rowMail === mailBuscado);
      } else if (!rowMail) {
        match = (rowNom === nameBuscado);
      }

      if (match) {
        sheet.getRange(i + 1, colPeso + 1).setValue(nuevoPeso);
        updatedRows++;
      }
    }

    return returnJSONP([{ ok: updatedRows > 0, updated: updatedRows, peso: nuevoPeso }], eForJSONP);

  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

function getAdminEmails() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Mascota Paseo") || ss.getSheets()[0];
  var data = sheet.getDataRange().getDisplayValues();

  var headerRowIndex = -1;
  for (var r = 0; r < Math.min(data.length, 10); r++) {
    for (var c = 0; c < data[r].length; c++) {
      var cell = String(data[r][c]).toLowerCase().trim();
      if (cell.indexOf('nombre') !== -1 || cell.indexOf('mascota') !== -1) {
        headerRowIndex = r; break;
      }
    }
    if (headerRowIndex !== -1) break;
  }
  if (headerRowIndex === -1) return [];

  var headers = data[headerRowIndex];
  var colMail = -1, colRol = -1;
  for (var j = 0; j < headers.length; j++) {
    var h = String(headers[j]).toLowerCase().trim()
      .normalize('NFD').replace(/[\u0300-\u036f]/g, '');
    if (h.indexOf('mail') !== -1 || h.indexOf('email') !== -1 || h.indexOf('correo') !== -1) colMail = j;
    if (h === 'rol' || h === 'role') colRol = j;
  }
  if (colMail === -1 || colRol === -1) return [];

  var emails = [];
  var seen = {};
  for (var i = headerRowIndex + 1; i < data.length; i++) {
    var row = data[i];
    var rol = String(row[colRol] || '').trim().toLowerCase()
      .normalize('NFD').replace(/[\u0300-\u036f]/g, '');
    if (rol !== 'admin') continue;
    var mail = String(row[colMail] || '').trim().toLowerCase();
    if (!mail || mail.indexOf('@') === -1) continue;
    if (seen[mail]) continue;
    seen[mail] = true;
    emails.push(mail);
  }
  return emails;
}

    if (!adminEmails.length) {
      return returnJSONP([{ ok: false, error: 'No se encontró ningún usuario con Rol=Admin en la hoja "Mascota Paseo"' }], eForJSONP);
    }

    var seccionKey = String(p.seccion || '').trim().toLowerCase().replace(/[\s_]/g, '');
    var seccionLabel = SECCION_LABEL[seccionKey] || String(p.seccion || 'Sección desconocida');

    var accion = String(p.accion || 'alta').trim().toLowerCase();
    var accionLabel = (accion === 'modificacion' || accion === 'edicion') ? 'modificó' : 'agregó';

    var userMail = String(p.userMail || '').trim().toLowerCase() || '(mail no disponible)';
    var petName = String(p.petName || '').trim() || '(mascota no especificada)';
    var detalle = String(p.detalle || '').trim();

    var asunto = 'Patas y Bigotes — ' + seccionLabel + ': ' + accionLabel + ' un registro';

    var cuerpo = 'El usuario ' + userMail + ' ' + accionLabel + ' un registro en la sección "' + seccionLabel + '".\n\n' +
      'Mascota: ' + petName + '\n' +
      'Sección: ' + seccionLabel + '\n' +
      'Acción: ' + (accion === 'modificacion' || accion === 'edicion' ? 'Modificación' : 'Alta nueva') + '\n';

    if (detalle) {
      cuerpo += 'Detalle: ' + detalle + '\n';
    }

    cuerpo += '\nEste es un mail automático generado por el sistema de Patas y Bigotes.';

    var enviados = 0;
    var errores = [];
    for (var k = 0; k < adminEmails.length; k++) {
      try {
        MailApp.sendEmail({
          to: adminEmails[k],
          subject: asunto,
          body: cuerpo
        });
        enviados++;
      } catch (mailErr) {
        errores.push(adminEmails[k] + ': ' + mailErr.toString());
      }
    }

    return returnJSONP([{ ok: enviados > 0, enviados: enviados, total: adminEmails.length, errores: errores }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

// ═══════════════════════════════════════════════════════════════
//  eliminarDuplicadosDesparasitarios / eliminarDuplicadosAlimentacion
//  — FIX v27, EJECUTAR UNA SOLA VEZ A MANO desde el editor de Apps
//  Script, para limpiar filas duplicadas que ya hayan quedado
//  cargadas en las hojas ANTES de este fix (el fix de
//  handleAddDesparasitario/handleAddAlimentacion evita duplicados
//  NUEVOS a partir de ahora, pero no borra los que ya están).
//
//  Recorre las filas con contenido en A-G (Desparasitarios) o A-J
//  (Alimentacion) y, para cada grupo de filas que comparta el mismo
//  Id, conserva solo la ÚLTIMA (la más reciente/más abajo en la hoja)
//  y limpia (clearContent, mismo criterio que handleDelete*) el resto
//  del bloque de registro de las demás -- así no se pierde ningún
//  dato real: si dos filas con el mismo id tenían pequeñas
//  diferencias (por ejemplo, una versión vieja de las fechas antes de
//  una edición), se conserva la más reciente, que es la que refleja
//  el estado real del registro.
//
//  Filas SIN Id (registros viejos previos a la columna Id) no se
//  tocan: no hay forma segura de saber si son duplicados entre sí sin
//  Id para compararlas.
//
//  Es seguro correrla de más: si no encuentra duplicados, no hace
//  ningún cambio y solo loguea "0 duplicados encontrados".
// ═══════════════════════════════════════════════════════════════
function eliminarDuplicadosDesparasitarios() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Desparasitarios');
  if (!sheet) {
    Logger.log('No existe la hoja "Desparasitarios" todavía; no hay nada que limpiar.');
    return;
  }
  var headerMap = getHeaderMap(sheet);
  if (!headerMap.hasOwnProperty('id')) {
    Logger.log('La hoja "Desparasitarios" no tiene columna Id; no se puede detectar duplicados de forma segura.');
    return;
  }
  var colId = headerMap['id'];
  var colsToClear = getRegistroWidth(sheet, COLUMNAS_REGISTRO_DESPARASITACION);
  var data = sheet.getDataRange().getDisplayValues();

  var lastRowByid = {}; // id -> número de fila (1-indexed) de la ÚLTIMA aparición
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var rowHasContent = false;
    for (var c = 0; c < Math.min(row.length, COLUMNAS_REGISTRO_DESPARASITACION); c++) {
      if (String(row[c]).trim() !== '') { rowHasContent = true; break; }
    }
    if (!rowHasContent) continue;
    var idVal = String(row[colId] || '').trim();
    if (!idVal) continue;
    lastRowByid[idVal] = i + 1; // se sobreescribe: la última aparición gana
  }

  var limpiadas = 0;
  for (var i2 = 1; i2 < data.length; i2++) {
    var row2 = data[i2];
    var rowHasContent2 = false;
    for (var c2 = 0; c2 < Math.min(row2.length, COLUMNAS_REGISTRO_DESPARASITACION); c2++) {
      if (String(row2[c2]).trim() !== '') { rowHasContent2 = true; break; }
    }
    if (!rowHasContent2) continue;
    var idVal2 = String(row2[colId] || '').trim();
    if (!idVal2) continue;

    var rowNumber2 = i2 + 1;
    var esLaUltima = lastRowByid[idVal2] === rowNumber2;
    if (!esLaUltima) {
      sheet.getRange(rowNumber2, 1, 1, colsToClear).clearContent();
      limpiadas++;
    }
  }

  Logger.log('Limpieza de Desparasitarios completa: ' + limpiadas + ' fila(s) duplicada(s) vaciada(s) (se conservó la más reciente de cada Id repetido).');
}

// ═══════════════════════════════════════════════════════════════
//  ██  REGISTRO Y VINCULACIÓN DE CLIENTES (secciones 6, 7 y 8)  ██
//
//  Criterio acordado: el alta automática solo procede cuando la
//  identidad es INEQUÍVOCA. Ante cualquier ambigüedad se crea el cliente
//  en estado PENDIENTE y la vinculación real la hace el administrador.
//
//  Reglas, en orden de prioridad:
//   A) El email verificado por Firebase ya figura en la columna Mail de
//      "Mascota Paseo" → vínculo directo (ACTIVO). Es inequívoco: ese
//      mail lo cargó el administrador y Firebase probó que la persona
//      controla esa casilla.
//   B) Nombre coincide con un Dueño Y el teléfono coincide con el que
//      figura en la planilla → ACTIVO.
//   C) Cualquier otro caso (nombre duplicado, sin teléfono con qué
//      desempatar, o sin coincidencias) → PENDIENTE.
//
//  ⚠️ Hoy "Mascota Paseo" NO tiene columna Teléfono, así que la regla B
//  no se puede evaluar y todo lo que no entre por A cae en PENDIENTE.
//  Es el comportamiento seguro. Si más adelante se agrega una columna
//  Telefono, la regla B empieza a funcionar sola, sin tocar el código.
// ═══════════════════════════════════════════════════════════════

function _norm(s) {
  return String(s || '').trim().toLowerCase()
    .normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/\s+/g, ' ');
}
function _normPhone(s) {
  return String(s || '').replace(/[^0-9]/g, '').replace(/^0+/, '');
}

// "Mascota Paseo" tiene los headers en la fila 4, no en la 1: hay que
// detectarlos en vez de asumir la primera fila.
function getMascotaPaseoInfo() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Mascota Paseo');
  if (!sheet) return null;
  var data = sheet.getDataRange().getDisplayValues();
  var hr = -1;
  for (var r = 0; r < Math.min(data.length, 10); r++) {
    for (var c = 0; c < data[r].length; c++) {
      var v = _norm(data[r][c]);
      if (v.indexOf('duen') !== -1 || v.indexOf('nombre mascota') !== -1) { hr = r; break; }
    }
    if (hr !== -1) break;
  }
  if (hr === -1) return null;
  var map = {};
  for (var c2 = 0; c2 < data[hr].length; c2++) {
    var n = _norm(data[hr][c2]).replace(/[\s_]/g, '');
    if (n && !map.hasOwnProperty(n)) map[n] = c2; // primera aparición gana
  }
  return { sheet: sheet, data: data, headerRow: hr, map: map };
}

function nextClienteId(sheet) {
  var data = sheet.getDataRange().getDisplayValues();
  var max = 0;
  for (var i = 1; i < data.length; i++) {
    var m = String(data[i][0] || '').match(/(\d+)\s*$/);
    if (m) { var n = parseInt(m[1], 10); if (n > max) max = n; }
  }
  var next = max + 1;
  return 'CLI-' + (next < 1000 ? ('000' + next).slice(-4) : String(next));
}

// ── Escribe CLIENTE_ID (y Mail si estaba vacío) en todas las hojas ──
function backfillClienteVinculo(clienteId, dueno, email) {
  var tocadas = { 'Mascota Paseo': 0, Alimentacion: 0, Desparasitarios: 0, CuidadoDomicilio: 0 };

  // 1) Mascota Paseo: por Dueño (ahí es donde vive la relación real).
  var info = getMascotaPaseoInfo();
  if (info && info.map.hasOwnProperty('dueno')) {
    var colDue = info.map['dueno'];
    var colMail = info.map.hasOwnProperty('mail') ? info.map['mail'] : -1;
    var colCid = info.map.hasOwnProperty('clienteid') ? info.map['clienteid'] : -1;
    for (var i = info.headerRow + 1; i < info.data.length; i++) {
      if (_norm(info.data[i][colDue]) !== _norm(dueno)) continue;
      if (colCid !== -1 && !String(info.data[i][colCid] || '').trim()) {
        info.sheet.getRange(i + 1, colCid + 1).setValue(clienteId);
        tocadas['Mascota Paseo']++;
      }
      // Nunca se inventa un email: solo se completa si está vacío.
      if (colMail !== -1 && email && !String(info.data[i][colMail] || '').trim()) {
        info.sheet.getRange(i + 1, colMail + 1).setValue(email);
      }
    }
  }

  // 2) Resto de hojas: por Mail (headers en fila 1).
  ['Alimentacion', 'Desparasitarios', 'CuidadoDomicilio'].forEach(function (nombreHoja) {
    if (!email) return;
    var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(nombreHoja);
    if (!sh) return;
    var d = sh.getDataRange().getDisplayValues();
    if (d.length < 2) return;
    var m = {};
    for (var c = 0; c < d[0].length; c++) {
      var n = _norm(d[0][c]).replace(/[\s_]/g, '');
      if (n && !m.hasOwnProperty(n)) m[n] = c;
    }
    if (!m.hasOwnProperty('mail') || !m.hasOwnProperty('clienteid')) return;
    for (var r = 1; r < d.length; r++) {
      if (_norm(d[r][m['mail']]) !== _norm(email)) continue;
      if (String(d[r][m['clienteid']] || '').trim()) continue;
      sh.getRange(r + 1, m['clienteid'] + 1).setValue(clienteId);
      tocadas[nombreHoja]++;
    }
  });

  return tocadas;
}

// ── Endpoint: alta de cliente tras registrarse en Firebase ──
function handleRegisterCliente(p, e) {
  try {
    var tok = verifyFirebaseToken(p.idToken || p.idtoken || p.token || '');
    if (!tok) return returnJSONP([{ ok: false, error: 'Token inválido o vencido.' }], e);

    var nombre = String(p.nombre || '').trim();
    var telefono = String(p.telefono || '').trim();
    if (!nombre) return returnJSONP([{ ok: false, error: 'Falta el nombre.' }], e);

    // Idempotente: si ese UID ya está registrado, se devuelve tal cual.
    var ya = resolveClienteFromUid(tok.uid);
    if (ya) {
      return returnJSONP([{ ok: true, yaRegistrado: true, clienteId: ya.clienteId, estado: ya.estado }], e);
    }

    var hoja = getClientesSheet();
    if (!hoja) return returnJSONP([{ ok: false, error: 'No existe la hoja Clientes.' }], e);

    var estado = 'PENDIENTE';
    var duenoVinculado = '';
    var info = getMascotaPaseoInfo();

    if (info && info.map.hasOwnProperty('dueno')) {
      var colDue = info.map['dueno'];
      var colMail = info.map.hasOwnProperty('mail') ? info.map['mail'] : -1;
      var colTel = info.map.hasOwnProperty('telefono') ? info.map['telefono']
        : (info.map.hasOwnProperty('tel') ? info.map['tel'] : -1);

      // ── Regla A: el email verificado ya está cargado en la planilla ──
      if (colMail !== -1) {
        for (var i = info.headerRow + 1; i < info.data.length; i++) {
          if (_norm(info.data[i][colMail]) === _norm(tok.email) && String(info.data[i][colDue] || '').trim()) {
            duenoVinculado = String(info.data[i][colDue]).trim();
            estado = 'ACTIVO';
            break;
          }
        }
      }

      // ── Regla B: nombre + teléfono coincidentes ──
      if (estado !== 'ACTIVO' && colTel !== -1 && telefono) {
        var candidatos = {};
        for (var j = info.headerRow + 1; j < info.data.length; j++) {
          var due = String(info.data[j][colDue] || '').trim();
          if (!due || _norm(due) !== _norm(nombre)) continue;
          if (_normPhone(info.data[j][colTel]) === _normPhone(telefono)) candidatos[_norm(due)] = due;
        }
        var claves = Object.keys(candidatos);
        // Un solo dueño candidato = inequívoco. Más de uno = ambiguo.
        if (claves.length === 1) { duenoVinculado = candidatos[claves[0]]; estado = 'ACTIVO'; }
      }
    }

    var clienteId = nextClienteId(hoja);
    var fecha = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');

    // Orden fijo según los headers de la hoja Clientes:
    // CLIENTE_ID | NOMBRE | EMAIL | TELEFONO | FECHA_ALTA | ESTADO | ROL | Firebase_UID
    hoja.appendRow([clienteId, nombre, tok.email, telefono, fecha, estado, 'usuario', tok.uid]);

    var tocadas = null;
    if (estado === 'ACTIVO') {
      tocadas = backfillClienteVinculo(clienteId, duenoVinculado, tok.email);
    }

    return returnJSONP([{
      ok: true,
      clienteId: clienteId,
      estado: estado,
      dueno: duenoVinculado,
      backfill: tocadas,
      mensaje: estado === 'ACTIVO'
        ? 'Cuenta vinculada correctamente.'
        : 'Necesitamos confirmar tus datos. Un administrador va a validar tu cuenta.'
    }], e);

  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], e);
  }
}

}

// ═══════════════════════════════════════════════════════════════
//  ██  MIGRACIÓN DE ESTRUCTURA — ejecutar UNA vez desde el editor  ██
//
//  Qué hace (es idempotente: se puede correr las veces que haga falta):
//   1) Agrega la columna CLIENTE_ID a las hojas que no la tengan,
//      insertándola JUSTO ANTES de la columna Id.
//
//      ¿Por qué antes de Id y no al final de la hoja, como decía el plan?
//      Porque las altas escriben solo el bloque de registro (A..Id, ancho
//      calculado por getRegistroWidth). Una CLIENTE_ID al final, después
//      del catálogo, quedaría fuera de ese bloque y NUNCA se completaría
//      en los registros nuevos. Insertarla antes de Id es seguro desde el
//      fix v23: getRegistroWidth se recalcula solo a partir del header.
//
//   2) Avisa si alguna hoja tiene CLIENTE_ID DUPLICADA (es el caso de
//      Desparasitarios, que la tiene en C y en I). Solo la elimina si la
//      sobrante está completamente vacía; si tiene datos, no toca nada y
//      lo reporta para revisarlo a mano.
// ═══════════════════════════════════════════════════════════════
function migrarEstructuraClienteId() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hojas = ['Alimentacion', 'Desparasitarios', 'CuidadoDomicilio', 'Mascota Paseo'];
  var reporte = [];

  hojas.forEach(function (nombre) {
    var sheet = ss.getSheetByName(nombre);
    if (!sheet) { reporte.push(nombre + ': no existe, se omite.'); return; }

    var data = sheet.getDataRange().getDisplayValues();
    if (!data.length) { reporte.push(nombre + ': vacía, se omite.'); return; }

    // Fila de headers (Mascota Paseo los tiene en la 4, no en la 1).
    var hr = 0;
    if (nombre === 'Mascota Paseo') {
      var info = getMascotaPaseoInfo();
      if (!info) { reporte.push(nombre + ': no se detectó la fila de headers.'); return; }
      hr = info.headerRow;
    }

    var cols = [];   // índices con header CLIENTE_ID
    var idCol = -1;
    for (var c = 0; c < data[hr].length; c++) {
      var n = _norm(data[hr][c]).replace(/[\s_]/g, '');
      if (n === 'clienteid') cols.push(c);
      if (n === 'id' && idCol === -1) idCol = c;
    }

    // ── Duplicadas ──
    if (cols.length > 1) {
      var sobrante = cols[cols.length - 1]; // se conserva la de más a la izquierda
      var vacia = true;
      for (var r = hr + 1; r < data.length; r++) {
        if (String(data[r][sobrante] || '').trim() !== '') { vacia = false; break; }
      }
      if (vacia) {
        sheet.deleteColumn(sobrante + 1);
        reporte.push(nombre + ': CLIENTE_ID duplicada en la columna ' + (sobrante + 1) + ' estaba vacía → eliminada.');
      } else {
        reporte.push('⚠️ ' + nombre + ': CLIENTE_ID duplicada en la columna ' + (sobrante + 1) +
          ' TIENE DATOS. No se tocó: revisala a mano antes de seguir.');
      }
      return;
    }

    if (cols.length === 1) { reporte.push(nombre + ': ya tiene CLIENTE_ID (columna ' + (cols[0] + 1) + ').'); return; }

    // ── No la tiene: se inserta antes de Id ──
    if (idCol === -1) { reporte.push('⚠️ ' + nombre + ': no se encontró columna Id; no se insertó CLIENTE_ID.'); return; }
    sheet.insertColumnBefore(idCol + 1);
    sheet.getRange(hr + 1, idCol + 1).setValue('CLIENTE_ID');
    reporte.push('✅ ' + nombre + ': CLIENTE_ID insertada en la columna ' + (idCol + 1) + ' (antes de Id).');
  });

  Logger.log(reporte.join('\n'));
  return reporte.join('\n');
}

function returnJSONP(rows, e) {
  var callback = e && e.parameter && e.parameter.callback;
  var json = JSON.stringify(rows);
  if (callback) {
    return ContentService
      .createTextOutput(callback + '(' + json + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(json)
    .setMimeType(ContentService.MimeType.JSON);
}

    .setMimeType(ContentService.MimeType.JSON);
}

// ═══════════════════════════════════════════════════════════════
//  registrarPago — sin cambios de comportamiento en v25 respecto a
//  v24: sigue aceptando las mismas 7 keys opcionales
//  (clienteId, firebaseUid, comprobanteUrl, comprobanteNombre,
//  estadoComprobante, fechaVerificacion, verificadoPor). Lo que
//  cambia en v25 es que, por primera vez, un llamador
//  (handleConfirmarPagoTransferencia, cuando le llegan esos datos
//  opcionales, y handleSubirComprobante indirectamente a través de la
//  URL que devuelve) puede llegar a pasarle un valor no vacío para
//  clienteId/comprobanteUrl/comprobanteNombre/estadoComprobante. Si
//  no vienen, sigue guardando '' exactamente igual que siempre.
//
//  appendRowByHeader (sin modificar) ya ignora cualquier key de
//  `values` que headerMap no reconozca, así que si la hoja real
//  todavía no tiene esas 7 columnas creadas a mano, estas 7 keys
//  simplemente no encuentran destino y no pasa nada.
// ═══════════════════════════════════════════════════════════════
function registrarPago(campos) {
  var sheet = getPagosSheet();
  var headerMap = getHeaderMap(sheet);
  var ref = String(campos.referenciaExterna || '').trim();
  var method = String(campos.metodo || '').trim();
  if (ref && headerMap.hasOwnProperty('referenciaexterna') && headerMap.hasOwnProperty('metodo')) {
    var existing = sheet.getDataRange().getDisplayValues();
    var refCol = headerMap['referenciaexterna'];
    var methodCol = headerMap['metodo'];
    var idCol = headerMap.hasOwnProperty('id') ? headerMap['id'] : -1;
    if (idCol !== -1) {
      for (var er = 1; er < existing.length; er++) {
        if (String(existing[er][refCol] || '').trim() === ref && String(existing[er][methodCol] || '').trim() === method) {
          return String(existing[er][idCol] || '').trim();
        }
      }
    }
  }
  var id = Utilities.getUuid();
  var fecha = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  var values = {
    'id': id,
    'fecha': fecha,
    'metodo': campos.metodo || '',
    'estado': campos.estado || '',
    'monto': campos.monto || 0,
    'descripcion': campos.descripcion || '',
    'mail': campos.mail || '',
    'referenciaexterna': campos.referenciaExterna || '',
    'mppaymentid': campos.mpPaymentId || '',
    'mpstatusdetail': campos.mpStatusDetail || '',
    'alias': campos.alias || '',
    // ── CLIENTE_ID y sistema de comprobantes ──
    'cliente_id': campos.clienteId || '',
    'firebase_uid': campos.firebaseUid || '',
    'comprobanteurl': campos.comprobanteUrl || '',
    'comprobantenombre': campos.comprobanteNombre || '',
    'estadocomprobante': campos.estadoComprobante || '',
    'fechaverificacion': campos.fechaVerificacion || '',
    'verificadopor': campos.verificadoPor || ''
  };
  appendRowByHeader(sheet, headerMap, values);
  return id;
}

// ── Access Token de Mercado Pago: SIEMPRE desde Script Properties.
//    Configuración del proyecto (⚙️) → Propiedades del script →
//    agregar MP_ACCESS_TOKEN con tu token (Production o Test, según
//    corresponda). Nunca hardcodear el Access Token acá; a diferencia
//    de la Public Key (que sí va en el HTML porque está pensada para
//    viajar al navegador), el Access Token puede crear/cancelar pagos
//    y tiene que quedar solo del lado del servidor. ──
function getMercadoPagoAccessToken() {
  return PropertiesService.getScriptProperties().getProperty('MP_ACCESS_TOKEN') || '';
}

// ═══════════════════════════════════════════════════════════════
//  crearPagoMercadoPago — recibe el token de tarjeta que el SDK de MP
//  ya generó EN EL NAVEGADOR (nunca ve el número de tarjeta real, solo
//  el token), y crea el pago vía la API de Payments de Mercado Pago:
//    POST https://api.mercadopago.com/v1/payments
//  Body de entrada esperado (lo manda fwHandleMercadoPagoSubmit del
//  frontend, sin tocar esa función):
//    { action, token, paymentMethodId, installments, issuerId,
//      amount, description, payerEmail, externalReference }
// ═══════════════════════════════════════════════════════════════
function handleCrearPagoMercadoPago(p, eForJSONP) {
  try {
    var accessToken = getMercadoPagoAccessToken();
    if (!accessToken) {
      return returnJSONP([{ ok: false, error: 'Falta configurar MP_ACCESS_TOKEN en Propiedades del script.' }], eForJSONP);
    }

    var token = String(p.token || '').trim();
    var monto = Number(p.amount) || 0;
    if (!token) return returnJSONP([{ ok: false, error: 'Falta el token de la tarjeta.' }], eForJSONP);
    if (monto <= 0) return returnJSONP([{ ok: false, error: 'Monto inválido.' }], eForJSONP);

    var installments = parseInt(p.installments, 10) || 1;

    var payload = {
      transaction_amount: monto,
      token: token,
      description: String(p.description || 'Compra Patas y Bigotes'),
      installments: installments,
      payment_method_id: String(p.paymentMethodId || ''),
      payer: { email: String(p.payerEmail || '').trim().toLowerCase() },
      external_reference: String(p.externalReference || '')
    };
    if (p.issuerId) payload.issuer_id = String(p.issuerId);

    var idempotencyKey = String(p.externalReference || '') + '-' + Utilities.getUuid();

    var res = UrlFetchApp.fetch('https://api.mercadopago.com/v1/payments', {
      method: 'post',
      contentType: 'application/json',
      headers: {
        'Authorization': 'Bearer ' + accessToken,
        'X-Idempotency-Key': idempotencyKey
      },
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    var code = res.getResponseCode();
    var body = JSON.parse(res.getContentText());

    if (code !== 200 && code !== 201) {
      var mensajeError = (body && (body.message || (body.cause && body.cause[0] && body.cause[0].description))) || 'El pago fue rechazado.';
      var ctxError = getAuthContext(eForJSONP || { parameter: p });
      registrarPago({
        metodo: 'Mercado Pago', estado: 'Rechazado', monto: monto,
        descripcion: p.description, mail: ctxError && ctxError.cliente ? ctxError.cliente.email : p.payerEmail,
        referenciaExterna: p.externalReference, mpStatusDetail: mensajeError,
        clienteId: ctxError && ctxError.cliente ? ctxError.cliente.clienteId : '',
        firebaseUid: ctxError ? ctxError.uid : ''
      });
      return returnJSONP([{ ok: false, error: mensajeError }], eForJSONP);
    }

    var estadoMp = body.status;
    var ctxMp = getAuthContext(eForJSONP || { parameter: p });
    registrarPago({
      metodo: 'Mercado Pago',
      estado: estadoMp === 'approved' ? 'Aprobado' : (estadoMp === 'in_process' ? 'En revisión' : 'Rechazado'),
      monto: monto, descripcion: p.description, mail: ctxMp && ctxMp.cliente ? ctxMp.cliente.email : p.payerEmail,
      referenciaExterna: p.externalReference,
      mpPaymentId: body.id, mpStatusDetail: body.status_detail || '',
      clienteId: ctxMp && ctxMp.cliente ? ctxMp.cliente.clienteId : '',
      firebaseUid: ctxMp ? ctxMp.uid : ''
    });

    if (estadoMp === 'rejected') {
      return returnJSONP([{ ok: false, error: 'El pago fue rechazado por el medio de pago (' + (body.status_detail || 'motivo no especificado') + ').' }], eForJSONP);
    }

    return returnJSONP([{
      ok: true,
      status: estadoMp,
      paymentId: body.id,
      amount: body.transaction_amount,
      dateApproved: body.date_approved || ''
    }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

// ═══════════════════════════════════════════════════════════════
//  confirmarPagoEfectivo — no hay pasarela: solo deja constancia de
//  que el cliente eligió pagar en efectivo. Estado siempre "Pendiente"
//  (alguien tiene que cobrar en persona y confirmarlo después a mano
//  en la hoja "Pagos" o desde un endpoint de administración que se
//  agregue más adelante — no existe todavía).
//
//  v23: devuelve returnJSONP([{...}], eForJSONP) en vez de
//  jsonOut({...}). El frontend llama a esta acción vía fetchJSONP
//  (fwPagarEnEfectivo en index.html), que agrega un
//  <script src="URL&callback=nombre_unico"> y espera a que ESE SCRIPT
//  llame a window[nombre_unico](datos). jsonOut() devuelve
//  Content-Type application/json con un objeto JS suelto, que el
//  navegador descarga como <script> pero que no es código ejecutable
//  válido — nunca llama al callback, y a los 14000ms el propio
//  fetchJSONP lo rechaza como "JSONP timeout" (el fetch nunca falla
//  del lado de la red, por eso no se ve como error de servidor).
//  returnJSONP(rows, e) lee el parámetro ?callback=... de la URL y
//  envuelve la respuesta como nombre_unico([{...}]), que sí es JS
//  ejecutable y dispara el callback de inmediato.
//
//  v25: sin cambios en esta función. Sigue sin mandar clienteId,
//  firebaseUid ni ningún campo de comprobante a registrarPago —
//  quedan en blanco en la hoja hasta que se implemente esa
//  funcionalidad para efectivo también (fuera del alcance de este
//  cambio, que se limita a transferencia).
// ═══════════════════════════════════════════════════════════════
function handleConfirmarPagoEfectivo(p, eForJSONP) {
  try {
    var monto = Number(p.amount) || 0;
    if (monto <= 0) return returnJSONP([{ ok: false, error: 'Monto inválido.' }], eForJSONP);

    var ctx = getAuthContext(eForJSONP || { parameter: p });
    var comprobanteId = registrarPago({
      metodo: 'Efectivo', estado: 'Pendiente', monto: monto,

      referenciaExterna: p.externalReference,
      clienteId: ctx && ctx.cliente ? ctx.cliente.clienteId : '',
      firebaseUid: ctx ? ctx.uid : ''
    });

    return returnJSONP([{ ok: true, comprobanteId: comprobanteId }], eForJSONP);
  } catch (err) {

  }
}

// ═══════════════════════════════════════════════════════════════
//  confirmarPagoTransferencia — mismo criterio que efectivo: se deja
//  constancia en estado "Pendiente" y se le devuelve al cliente el
//  alias/CBU para que transfiera. El alias sale de Script Properties
//  (TRANSFERENCIA_ALIAS) — no se hardcodea en el código.
//
//  v23: returnJSONP([{...}], eForJSONP) en vez de jsonOut({...}), por
//  la misma razón que confirmarPagoEfectivo (llamada vía
//  fwPagarPorTransferencia/fetchJSONP).
//
//  v25: se agregan 4 parámetros OPCIONALES —comprobanteUrl,
//  comprobanteNombre, clienteId, firebaseUid— que, SI vienen, se pasan
//  a registrarPago junto con estadoComprobante:
//  'Pendiente de verificación'. Si no vienen (el caso normal: en el
//  momento de pedir el alias el cliente todavía no transfirió, así
//  que todavía no puede tener comprobante), se guardan como '' —
//  EXACTAMENTE el mismo comportamiento que v23/v24. El chequeo de
//  monto>0 sigue siendo la única validación de entrada; NO se agregó
//  ningún chequeo que exija comprobante, porque esta llamada ocurre
//  ANTES de que el comprobante pueda existir.
// ═══════════════════════════════════════════════════════════════
function handleConfirmarPagoTransferencia(p, eForJSONP) {
  try {
    var monto = Number(p.amount) || 0;
    if (monto <= 0) return returnJSONP([{ ok: false, error: 'Monto inválido.' }], eForJSONP);

    var alias = PropertiesService.getScriptProperties().getProperty('TRANSFERENCIA_ALIAS') || '';

    var comprobanteUrl = String(p.comprobanteUrl || '').trim();
    var comprobanteNombre = String(p.comprobanteNombre || '').trim();

    // La identidad NO sale del request del navegador. Se deriva del token
    // verificado por checkWriteAuth/getAuthContext; así un cliente no puede
    // escribir el CLIENTE_ID/Firebase_UID de otra persona en Pagos.
    var ctx = getAuthContext(eForJSONP || { parameter: p });
    var comprobanteId = registrarPago({
      metodo: 'Transferencia', estado: 'Pendiente', monto: monto,
      descripcion: p.description, mail: ctx && ctx.cliente ? ctx.cliente.email : p.payerEmail,
      referenciaExterna: p.externalReference, alias: alias,
      clienteId: ctx && ctx.cliente ? ctx.cliente.clienteId : '',
      firebaseUid: ctx ? ctx.uid : '',
      comprobanteUrl: comprobanteUrl,
      comprobanteNombre: comprobanteNombre,
      estadoComprobante: comprobanteUrl ? 'Pendiente de verificación' : ''
    });

    return returnJSONP([{
      ok: true, comprobanteId: comprobanteId, alias: alias,
    }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

function handleSubirComprobante(p, eForJSONP) {
  try {
    var base64Raw = String(p.comprobanteBase64 || '').trim();
    if (!base64Raw) {
      return returnJSONP([{ ok: false, error: 'Falta el archivo del comprobante (comprobanteBase64).' }], eForJSONP);
    }

    // Si el frontend manda el data URL completo
    // ("data:image/jpeg;base64,/9j/4AAQ..."), se recorta el prefijo
    // antes de decodificar; si ya manda solo el base64 puro, este
    // replace no encuentra la coma y no cambia nada.
    var base64Data = base64Raw;
    var commaIdx = base64Raw.indexOf(',');
    if (base64Raw.substring(0, 5) === 'data:' && commaIdx !== -1) {
      base64Data = base64Raw.substring(commaIdx + 1);
    }

    var nombreSugerido = String(p.comprobanteNombre || '').trim();
    var identificador = String(p.clienteId || p.mail || p.externalReference || 'sin-id').trim();
    var fechaArchivo = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd_HHmmss');
    var nombreFinal = nombreSugerido
      ? (fechaArchivo + '_' + identificador + '_' + nombreSugerido)
      : (fechaArchivo + '_' + identificador + '_comprobante.jpg');

    var bytes;
    try {
      bytes = Utilities.base64Decode(base64Data);
    } catch (decErr) {
      return returnJSONP([{ ok: false, error: 'El archivo no es un base64 válido.' }], eForJSONP);
    }

    // MimeType: se infiere del nombre si trae extensión reconocible;
    // por defecto JPEG (caso más común de foto de comprobante sacada
    // con el celular).
    var mimeType = MimeType.JPEG;
    var nombreLower = nombreFinal.toLowerCase();
    if (nombreLower.indexOf('.png') !== -1) mimeType = MimeType.PNG;
    else if (nombreLower.indexOf('.pdf') !== -1) mimeType = MimeType.PDF;

    var blob = Utilities.newBlob(bytes, mimeType, nombreFinal);

    var folder = getComprobantesFolder();
    var file = folder.createFile(blob);
    // Solo lectura para quien tenga el link (necesario para que el
    // link funcione al mostrarlo en el portal/administración sin
    // pedir login de Drive); nunca "cualquiera puede editar".
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var fileId = file.getId();
    var url = 'https://lh3.googleusercontent.com/d/' + fileId; // mismo formato que ya usa extractPhotoUrl para fotos

    return returnJSONP([{ ok: true, url: url, fileId: fileId, nombre: nombreFinal }], eForJSONP);
  } catch (err) {
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

function handleCrearPreferenciaMP(p, eForJSONP) {
  try {
    var accessToken = PropertiesService.getScriptProperties().getProperty('MP_ACCESS_TOKEN');
    if (!accessToken) {
      throw new Error('Falta configurar MP_ACCESS_TOKEN en las Propiedades del Script.');
    }

    var amount = Number(p.amount || p.monto || 100);
    var title = p.description || p.title || p.descripcion || 'Pago Patas y Bigotes';
    var payerEmail = p.payerEmail || p.mail || '';
    var externalReference = String(p.externalReference || '').trim();

    var payload = {
      items: [
        {
          title: title,
          description: title,
          unit_price: amount,
          quantity: 1,
          currency_id: 'ARS'
        }
      ],
      payer: payerEmail ? { email: payerEmail } : undefined,
      external_reference: externalReference,
      back_urls: {
        success: 'https://patas-y-bigotes.vercel.app',
        failure: 'https://patas-y-bigotes.vercel.app',
        pending: 'https://patas-y-bigotes.vercel.app'
      },
      auto_return: 'approved'
    };

    var options = {
      method: 'post',
      contentType: 'application/json',
      headers: {
        'Authorization': 'Bearer ' + accessToken
      },
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    };

    var response = UrlFetchApp.fetch('https://api.mercadopago.com/checkout/preferences', options);
    var json = JSON.parse(response.getContentText());
    Logger.log('MP_ACCESS_TOKEN presente: ' + (!!accessToken));
    Logger.log('Payload enviado: ' + JSON.stringify(payload));
    Logger.log('Response code: ' + response.getResponseCode());
    Logger.log('Response body: ' + response.getContentText());

    if (response.getResponseCode() !== 201 && response.getResponseCode() !== 200) {
      // NUEVO: log explícito del motivo antes de tirar el error, para
      // no tener que reconstruirlo a mano desde el log crudo de arriba.
      Logger.log('[crearPreferenciaMP] RECHAZADO por Mercado Pago. message=' +
        (json && json.message) + ' | cause=' + (json && json.cause ? JSON.stringify(json.cause) : '(sin cause)'));
      throw new Error('Error MP: ' + (json.message || response.getContentText()));
    }

    // FIX: array, no objeto suelto -- mismo contrato que las otras 3
    // funciones de pago (crearPagoMercadoPago, confirmarPagoEfectivo,
    // confirmarPagoTransferencia).
    return returnJSONP([{ ok: true, preferenceId: json.id }], eForJSONP);
  } catch (err) {
    Logger.log('[crearPreferenciaMP] EXCEPCIÓN: ' + err.toString());
    // FIX: mismo cambio, array en vez de objeto suelto.
    return returnJSONP([{ ok: false, error: err.toString() }], eForJSONP);
  }
}

// ── Helper interno: exige admin verificado por token, o devuelve el
//    mensaje de error para responder directo. null = autorizado. ──
function _requireAdminCtx(e) {
  var ctx = getAuthContext(e);
  if (ctx.mode === 'denied') return { ctx: null, error: 'No autorizado: falta un idToken válido.' };
  if (ctx.mode === 'legacy') return { ctx: null, error: 'No autorizado: esta acción requiere autenticación.' };
  if (!ctx.isAdmin) return { ctx: null, error: 'No autorizado: se requiere rol de administrador.' };
  return { ctx: ctx, error: null };
}

var AGENDA_PASEOS_SHEET = 'Mascota Paseo';
var AGENDA_HISTORIAL_SHEET = 'Historial_Paseos';

function normalizePaseoKey_(s) {
  return String(s || '').trim().toLowerCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .replace(/[\s_]+/g, '');
}

function normalizePaseoDay_(s) {
  var d = normalizePaseoKey_(s);
  if (!d) return '';
  if (d.indexOf('lun') === 0) return 'Lunes';
  if (d.indexOf('mar') === 0) return 'Martes';
  if (d.indexOf('mie') === 0) return 'Miércoles';
  if (d.indexOf('jue') === 0) return 'Jueves';
  if (d.indexOf('vie') === 0) return 'Viernes';
  if (d.indexOf('sab') === 0) return 'Sábado';
  if (d.indexOf('dom') === 0) return 'Domingo';
  return String(s || '').trim();
}

function normalizePaseoTime_(s) {
  var v = String(s || '').trim().replace(/\s*(hs|hrs|h)\s*$/i, '');
  if (!v) return '';
  var m = v.match(/^(\d{1,2}):(\d{2})/);
  if (m) return m[1].padStart(2,'0') + ':' + m[2];
  if (/^\d{1,2}$/.test(v)) return v.padStart(2,'0') + ':00';
  return v;
}

function getMascotaPaseoHeaderContext_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(AGENDA_PASEOS_SHEET);
  if (!sheet) throw new Error('No existe la hoja "' + AGENDA_PASEOS_SHEET + '".');

  var maxScanRows = Math.min(Math.max(sheet.getLastRow(), 1), 10);
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var scan = sheet.getRange(1, 1, maxScanRows, lastCol).getDisplayValues();

  var headerRow = -1;
  for (var r = 0; r < scan.length; r++) {
    var normalized = scan[r].map(normalizePaseoKey_);
    var hasPet = normalized.some(function(x) {
      return x === 'nombremascota' || x === 'mascota' || x === 'nombre';
    });
    var hasOwner = normalized.some(function(x) {
      return x === 'dueno' || x === 'owner' || x === 'propietario';
    });
    var hasCode = normalized.some(function(x) {
      return x === 'codigomascota';
    });
    if ((hasPet && hasOwner) || hasCode) {
      headerRow = r + 1;
      break;
    }
  }
  if (headerRow === -1) {
    throw new Error('No pude identificar la fila de encabezados de "' + AGENDA_PASEOS_SHEET + '".');
  }

  var headers = sheet.getRange(headerRow, 1, 1, lastCol).getDisplayValues()[0];
  var map = {};
  for (var c = 0; c < headers.length; c++) {
    var key = normalizePaseoKey_(headers[c]);
    if (key && !map.hasOwnProperty(key)) map[key] = c + 1; // 1-based
  }

  return { sheet: sheet, headerRow: headerRow, headers: headers, map: map };
}

function ensurePaseoIdColumn_() {
  var ctx = getMascotaPaseoHeaderContext_();
  var sheet = ctx.sheet;
  var col = ctx.map['paseoid'] || 0;

  if (!col) {
    col = sheet.getLastColumn() + 1;
    sheet.getRange(ctx.headerRow, col).setValue('PASEO_ID');
    ctx.map['paseoid'] = col;
  }

  var lastRow = sheet.getLastRow();
  if (lastRow <= ctx.headerRow) return {ctx: ctx, paseoCol: col};

  var lastCol = sheet.getLastColumn();
  var values = sheet.getRange(ctx.headerRow + 1, 1, lastRow - ctx.headerRow, lastCol).getDisplayValues();
  var updates = [];

  for (var i = 0; i < values.length; i++) {
    var row = values[i];
    var hasContent = false;
    for (var j = 0; j < row.length; j++) {
      if (j + 1 === col) continue;
      if (String(row[j] || '').trim() !== '') {
        hasContent = true;
        break;
      }
    }
    if (hasContent && !String(row[col - 1] || '').trim()) {
      updates.push({
        row: ctx.headerRow + 1 + i,
        value: Utilities.getUuid()
      });
    }
  }

  for (var u = 0; u < updates.length; u++) {
    sheet.getRange(updates[u].row, col).setValue(updates[u].value);
  }

  return {ctx: ctx, paseoCol: col};
}

function snapshotPaseoRow_(ctx, rowNumber) {
  var row = ctx.sheet.getRange(rowNumber, 1, 1, ctx.sheet.getLastColumn()).getDisplayValues()[0];

  function val() {
    for (var i = 0; i < arguments.length; i++) {
      var key = normalizePaseoKey_(arguments[i]);
      if (ctx.map.hasOwnProperty(key)) {
        return String(row[ctx.map[key] - 1] || '').trim();
      }
    }
    return '';
  }

  return {
    paseoId: val('PASEO_ID','PASEOID'),
    codigoMascota: val('Codigo Mascota','Código Mascota','Codigo_Mascota'),
    owner: val('Dueño','Dueno','Owner','Propietario'),
    nombreMascota: val('Nombre Mascota','Mascota','Nombre'),
    barrio: val('Barrio','Zona'),
    lote: val('Lote'),
    sexo: val('Sexo','Genero'),
    raza: val('Raza'),
    cumple: val('Cumple','Nacimiento'),
    pesoActual: val('Peso Actual','Peso'),
    dia: normalizePaseoDay_(val('Dia','Día')),
    horaInicio: normalizePaseoTime_(val('Hora inicio','Hora Inicio','Inicio')),
    horaFin: normalizePaseoTime_(val('Hora Fin','Hora fin','Fin')),
    cantDias: val('Cant dias','Cant días','Cantidad dias'),
    costo: parsePrecioCLP(val('Costo','Precio','Tarifa','Valor')),
    semanalAprox: parsePrecioCLP(val('R+ Aprox Semanal','R Aprox Semanal','Aprox Semanal')),
    mail: val('Mail','Email','Correo'),
    rol: val('Rol','Role'),
    serviciosActivos: val('Servicios_Activos','Servicios Activos','Servicios'),
    fotosBanners: val('Fotos_Banners','Fotos Banners','Foto'),
    clienteId: val('CLIENTE_ID','Cliente ID'),
    rowNumber: rowNumber
  };
}

function findPaseoRowById_(ctx, id) {
  var wanted = String(id || '').trim();
  if (!wanted || !ctx.map['paseoid']) return -1;

  var lastRow = ctx.sheet.getLastRow();
  if (lastRow <= ctx.headerRow) return -1;

  var col = ctx.map['paseoid'];
  var vals = ctx.sheet.getRange(ctx.headerRow + 1, col, lastRow - ctx.headerRow, 1).getDisplayValues();
  for (var i = 0; i < vals.length; i++) {
    if (String(vals[i][0] || '').trim() === wanted) {
      return ctx.headerRow + 1 + i;
    }
  }
  return -1;
}

function findDogTemplateRow_(ctx, p) {
  var lastRow = ctx.sheet.getLastRow();
  if (lastRow <= ctx.headerRow) return -1;

  var lastCol = ctx.sheet.getLastColumn();
  var data = ctx.sheet.getRange(ctx.headerRow + 1, 1, lastRow - ctx.headerRow, lastCol).getDisplayValues();

  var code = String(p.codigoMascota || p.codigo || p.codigo_mascota || '').trim();
  var pet = String(p.nombreMascota || p.petName || p.name || '').trim().toLowerCase();
  var owner = String(p.dueno || p.owner || '').trim().toLowerCase();

  var codeCol = ctx.map['codigomascota'] || 0;
  var petCol = ctx.map['nombremascota'] || ctx.map['mascota'] || ctx.map['nombre'] || 0;
  var ownerCol = ctx.map['dueno'] || ctx.map['owner'] || ctx.map['propietario'] || 0;

  if (code && codeCol) {
    for (var i = 0; i < data.length; i++) {
      if (String(data[i][codeCol - 1] || '').trim() === code) {
        return ctx.headerRow + 1 + i;
      }
    }
  }

  if (petCol && pet) {
    var candidates = [];
    for (var j = 0; j < data.length; j++) {
      var petVal = String(data[j][petCol - 1] || '').trim().toLowerCase();
      if (!petVal || petVal !== pet) continue;

      if (ownerCol && owner) {
        var ownerVal = String(data[j][ownerCol - 1] || '').trim().toLowerCase();
        if (ownerVal === owner) candidates.push(ctx.headerRow + 1 + j);
      } else if (!owner) {
        candidates.push(ctx.headerRow + 1 + j);
      }
    }
    if (candidates.length === 1) return candidates[0];
  }

  return -1;
}

function writePaseoSnapshotToRow_(ctx, rowNumber, snapshot) {
  var sheet = ctx.sheet;
  var map = ctx.map;

  function put(candidates, value) {
    for (var i = 0; i < candidates.length; i++) {
      var key = normalizePaseoKey_(candidates[i]);
      if (map.hasOwnProperty(key)) {
        sheet.getRange(rowNumber, map[key]).setValue(value);
        return;
      }
    }
  }

  put(['Codigo Mascota','Código Mascota','Codigo_Mascota'], snapshot.codigoMascota);
  put(['Dueño','Dueno','Owner','Propietario'], snapshot.owner);
  put(['Nombre Mascota','Mascota','Nombre'], snapshot.nombreMascota);
  put(['Barrio','Zona'], snapshot.barrio);
  put(['Lote'], snapshot.lote);
  put(['Sexo','Genero'], snapshot.sexo);
  put(['Raza'], snapshot.raza);
  put(['Cumple','Nacimiento'], snapshot.cumple);
  put(['Peso Actual','Peso'], snapshot.pesoActual);
  put(['Dia','Día'], snapshot.dia);
  put(['Hora inicio','Hora Inicio','Inicio'], snapshot.horaInicio);
  put(['Hora Fin','Hora fin','Fin'], snapshot.horaFin);
  put(['Cant dias','Cant días','Cantidad dias'], snapshot.cantDias || '1');
  put(['Costo','Precio','Tarifa','Valor'], snapshot.costo);
  put(['R+ Aprox Semanal','R Aprox Semanal','Aprox Semanal'], snapshot.semanalAprox || snapshot.costo);
  put(['Mail','Email','Correo'], snapshot.mail);
  put(['Rol','Role'], snapshot.rol || 'usuario');
  put(['Servicios_Activos','Servicios Activos','Servicios'], snapshot.serviciosActivos);
  put(['Fotos_Banners','Fotos Banners','Foto'], snapshot.fotosBanners);
  put(['CLIENTE_ID','Cliente ID'], snapshot.clienteId);
  put(['PASEO_ID'], snapshot.paseoId);

  var costoKey = normalizePaseoKey_('Costo');
  if (map.hasOwnProperty(costoKey)) {
    sheet.getRange(rowNumber, map[costoKey]).setNumberFormat('$ #,##0');
  }
}

function rowDataToSnapshotFromTemplate_(ctx, templateRow, p, newId) {
  var snap = snapshotPaseoRow_(ctx, templateRow);
  snap.paseoId = newId;

  if (p.codigoMascota || p.codigo || p.codigo_mascota) snap.codigoMascota = String(p.codigoMascota || p.codigo || p.codigo_mascota).trim();
  if (p.dueno || p.owner) snap.owner = String(p.dueno || p.owner).trim();
  if (p.nombreMascota || p.petName || p.name) snap.nombreMascota = String(p.nombreMascota || p.petName || p.name).trim();
  if (p.barrio || p.zona) snap.barrio = String(p.barrio || p.zona).trim();
  if (p.lote) snap.lote = String(p.lote).trim();
  if (p.sexo) snap.sexo = String(p.sexo).trim();
  if (p.raza) snap.raza = String(p.raza).trim();
  if (p.cumple || p.nacimiento) snap.cumple = String(p.cumple || p.nacimiento).trim();
  if (p.peso != null || p.pesoActual != null) snap.pesoActual = String(p.peso != null ? p.peso : p.pesoActual).trim();
  if (p.mail || p.email) snap.mail = String(p.mail || p.email).trim().toLowerCase();
  if (p.rol) snap.rol = String(p.rol).trim().toLowerCase();
  if (p.serviciosActivos || p.servicios) snap.serviciosActivos = String(p.serviciosActivos || p.servicios).trim();
  if (p.fotosBanners || p.foto) snap.fotosBanners = String(p.fotosBanners || p.foto).trim();
  if (p.clienteId) snap.clienteId = String(p.clienteId).trim();

  snap.dia = normalizePaseoDay_(p.day || p.dia || snap.dia);
  snap.horaInicio = normalizePaseoTime_(p.start || p.inicio || snap.horaInicio);
  snap.horaFin = normalizePaseoTime_(p.end || p.fin || snap.horaFin);
  snap.cantDias = String(p.cantDias || p.cantidadDias || snap.cantDias || '1').trim();

  var costoRaw = p.costo != null ? p.costo : (p.cost != null ? p.cost : '');
  if (String(costoRaw).trim() !== '') snap.costo = parsePrecioCLP(costoRaw);
  if (!snap.semanalAprox || Number(snap.semanalAprox) === 0) snap.semanalAprox = snap.costo;
  return snap;
}

function writeHistoryPaseo_(action, actorEmail, before, after, extra) {
  var sheet = getOrCreateSheet(AGENDA_HISTORIAL_SHEET, [
    'FechaHora','PASEO_ID','Accion','Usuario','Codigo Mascota','Dueño','Mascota',
    'Dia','Hora inicio','Hora Fin','Costo','Antes_JSON','Despues_JSON','Detalle'
  ]);
  var now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  var base = after || before || {};

  sheet.appendRow([
    now,
    base.paseoId || (before && before.paseoId) || '',
    action,
    actorEmail || '',
    base.codigoMascota || '',
    base.owner || '',
    base.nombreMascota || '',
    base.dia || '',
    base.horaInicio || '',
    base.horaFin || '',
    Number(base.costo || 0),
    before ? JSON.stringify(before) : '',
    after ? JSON.stringify(after) : '',
    extra || ''
  ]);
}

function handleAddPaseo(p, eForJSONP) {
  try {
    var auth = _requireAgendaAdmin_(eForJSONP || {parameter:p});
    if (auth.error) return returnJSONP([{ok:false,error:auth.error}], eForJSONP);

    var base = ensurePaseoIdColumn_();
    var ctx = base.ctx;
    var sheet = ctx.sheet;

    var templateRow = findDogTemplateRow_(ctx, p);
    if (templateRow === -1) {
      return returnJSONP([{ok:false,error:'No encontré la mascota para ese Código Mascota / Nombre + Dueño.'}], eForJSONP);
    }

    var newId = String(p.paseoId || Utilities.getUuid()).trim();
    var existingById = findPaseoRowById_(ctx, newId);
    if (existingById !== -1) {
      var existingSnap = snapshotPaseoRow_(ctx, existingById);
      return returnJSONP([{ok:true,created:false,id:newId,rowNumber:existingById,paseo:existingSnap}], eForJSONP);
    }

    var newSnap = rowDataToSnapshotFromTemplate_(ctx, templateRow, p, newId);

    // Evita duplicar involuntariamente el mismo paseo.
    var lastRow = sheet.getLastRow();
    if (lastRow > ctx.headerRow) {
      for (var r = ctx.headerRow + 1; r <= lastRow; r++) {
        var s = snapshotPaseoRow_(ctx, r);
        if (!s.nombreMascota) continue;

        var samePet = newSnap.codigoMascota && s.codigoMascota
          ? s.codigoMascota === newSnap.codigoMascota
          : (s.nombreMascota.toLowerCase() === newSnap.nombreMascota.toLowerCase() &&
             s.owner.toLowerCase() === newSnap.owner.toLowerCase());

        if (samePet &&
            s.dia === newSnap.dia &&
            s.horaInicio === newSnap.horaInicio &&
            s.horaFin === newSnap.horaFin) {
          return returnJSONP([{
            ok:true,created:false,duplicate:true,
            id:s.paseoId,rowNumber:r,paseo:s
          }], eForJSONP);
        }
      }
    }

    var targetRow = sheet.getLastRow() + 1;
    // Copiamos la fila plantilla completa antes de sobreescribir los campos
    // controlados. Así se conservan formato, fórmulas o columnas futuras
    // que no formen parte del snapshot.
    sheet.getRange(templateRow, 1, 1, sheet.getLastColumn()).copyTo(
      sheet.getRange(targetRow, 1, 1, sheet.getLastColumn()),
      SpreadsheetApp.CopyPasteType.PASTE_NORMAL,
      false
    );
    writePaseoSnapshotToRow_(ctx, targetRow, newSnap);
    var after = snapshotPaseoRow_(ctx, targetRow);
    writeHistoryPaseo_('ALTA', auth.ctx.cliente.email, null, after, '');

    return returnJSONP([{ok:true,created:true,id:newId,rowNumber:targetRow,paseo:after}], eForJSONP);
  } catch (err) {
    return returnJSONP([{ok:false,error:err.toString()}], eForJSONP);
  }
}

function handleUpdatePaseo(p, eForJSONP) {
  try {
    var auth = _requireAgendaAdmin_(eForJSONP || {parameter:p});
    if (auth.error) return returnJSONP([{ok:false,error:auth.error}], eForJSONP);

    var base = ensurePaseoIdColumn_();
    var ctx = base.ctx;
    var id = String(p.paseoId || p.id || '').trim();
    if (!id) return returnJSONP([{ok:false,error:'Falta PASEO_ID.'}], eForJSONP);

    var rowNumber = findPaseoRowById_(ctx, id);
    if (rowNumber === -1) return returnJSONP([{ok:false,error:'No existe un paseo con ese PASEO_ID.'}], eForJSONP);

    var before = snapshotPaseoRow_(ctx, rowNumber);
    var updated = JSON.parse(JSON.stringify(before));

    // Si cambia de mascota, copiamos el bloque estable de la mascota destino.
    var hasDogChange = !!(p.codigoMascota || p.codigo || p.codigo_mascota || p.nombreMascota || p.petName || p.name || p.owner || p.dueno);
    if (hasDogChange) {
      var templateRow = findDogTemplateRow_(ctx, p);
      if (templateRow === -1) return returnJSONP([{ok:false,error:'No encontré la mascota destino para la actualización.'}], eForJSONP);

      var template = snapshotPaseoRow_(ctx, templateRow);
      updated.codigoMascota = template.codigoMascota;
      updated.owner = template.owner;
      updated.nombreMascota = template.nombreMascota;
      updated.barrio = template.barrio;
      updated.lote = template.lote;
      updated.sexo = template.sexo;
      updated.raza = template.raza;
      updated.cumple = template.cumple;
      updated.pesoActual = template.pesoActual;
      updated.mail = template.mail;
      updated.rol = template.rol;
      updated.serviciosActivos = template.serviciosActivos;
      updated.fotosBanners = template.fotosBanners;
      updated.clienteId = template.clienteId;
    }

    if (p.day || p.dia) updated.dia = normalizePaseoDay_(p.day || p.dia);
    if (p.start || p.inicio) updated.horaInicio = normalizePaseoTime_(p.start || p.inicio);
    if (p.end || p.fin) updated.horaFin = normalizePaseoTime_(p.end || p.fin);
    if (p.cantDias || p.cantidadDias) updated.cantDias = String(p.cantDias || p.cantidadDias).trim();
    if (p.cost != null || p.costo != null) updated.costo = parsePrecioCLP(p.cost != null ? p.cost : p.costo);
    updated.paseoId = before.paseoId;

    writePaseoSnapshotToRow_(ctx, rowNumber, updated);
    var after = snapshotPaseoRow_(ctx, rowNumber);
    writeHistoryPaseo_('EDICION', auth.ctx.cliente.email, before, after, '');

    return returnJSONP([{ok:true,id:id,rowNumber:rowNumber,paseo:after}], eForJSONP);
  } catch (err) {
    return returnJSONP([{ok:false,error:err.toString()}], eForJSONP);
  }
}

function handleDuplicatePaseo(p, eForJSONP) {
  try {
    var auth = _requireAgendaAdmin_(eForJSONP || {parameter:p});
    if (auth.error) return returnJSONP([{ok:false,error:auth.error}], eForJSONP);

    var base = ensurePaseoIdColumn_();
    var ctx = base.ctx;
    var id = String(p.paseoId || p.id || '').trim();
    if (!id) return returnJSONP([{ok:false,error:'Falta PASEO_ID.'}], eForJSONP);

    var sourceRow = findPaseoRowById_(ctx, id);
    if (sourceRow === -1) return returnJSONP([{ok:false,error:'No existe un paseo con ese PASEO_ID.'}], eForJSONP);

    var source = snapshotPaseoRow_(ctx, sourceRow);
    var copy = JSON.parse(JSON.stringify(source));
    var newId = Utilities.getUuid();
    copy.paseoId = newId;

    if (p.day || p.dia) copy.dia = normalizePaseoDay_(p.day || p.dia);
    if (p.start || p.inicio) copy.horaInicio = normalizePaseoTime_(p.start || p.inicio);
    if (p.end || p.fin) copy.horaFin = normalizePaseoTime_(p.end || p.fin);

    var targetRow = ctx.sheet.getLastRow() + 1;
    ctx.sheet.getRange(sourceRow, 1, 1, ctx.sheet.getLastColumn()).copyTo(
      ctx.sheet.getRange(targetRow, 1, 1, ctx.sheet.getLastColumn()),
      SpreadsheetApp.CopyPasteType.PASTE_NORMAL,
      false
    );
    writePaseoSnapshotToRow_(ctx, targetRow, copy);
    var after = snapshotPaseoRow_(ctx, targetRow);
    writeHistoryPaseo_('DUPLICACION', auth.ctx.cliente.email, source, after, 'Nuevo PASEO_ID=' + newId);

    return returnJSONP([{ok:true,id:newId,sourceId:id,rowNumber:targetRow,paseo:after}], eForJSONP);
  } catch (err) {
    return returnJSONP([{ok:false,error:err.toString()}], eForJSONP);
  }
}

function handleGetHistorialPaseo(e) {
  try {
    var auth = _requireAgendaAdmin_(e || {parameter:{}});
    if (auth.error) return returnJSONP([{ok:false,error:auth.error}], e);

    var p = (e && e.parameter) ? e.parameter : {};
    var id = String(p.paseoId || p.id || '').trim();

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(AGENDA_HISTORIAL_SHEET);
    if (!sheet) return returnJSONP([], e);

    var rows = sheetToObjects(sheet);
    if (id) {
      rows = rows.filter(function(r) {
        return String(r['PASEO_ID'] || r['Paseo_ID'] || '').trim() === id;
      });
    }
    return returnJSONP(rows, e);
  } catch (err) {
    return returnJSONP([{ok:false,error:err.toString()}], e);
  }
}
